## Improved updateNextPickDisplay
```javascript
function updateNextPickDisplay() {
  // find the user's next pick index
  let nextPos = -1;
  for (let i = draftSim.pos; i < draftSim.order.length; i++) {
    if (draftSim.order[i] === draftSim.userIndex) { nextPos = i; break; }
  }
  const badge = document.getElementById('nextPickBadge');
  if (!badge) return;
  if (nextPos < 0) { badge.textContent = ''; return; }
  const round = Math.floor(nextPos / 32) + 1;
  const pickInRound = (nextPos % 32) + 1;
  badge.textContent = `(${round}.${pickInRound})`;
}
```

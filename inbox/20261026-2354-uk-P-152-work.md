### 1. New Card HTML Structure

```javascript
function renderPlayerCard(r, g, mx, sim, roster, draftUI){
  const p = r.p;
  const name = p[0];
  const pos = p[1];
  const team = p[2];
  const ovr = p[3];
  const teamAbbr = TEAM_ABBR[team] || team;
  const teamColor = getTeamColor(team);
  const teamName = team; // Full team name
  
  // OVR bubbles (only for 85+)
  const ovrBubbles = ovr >= 85 ? `
    <div class="pcard-ovr-bubbles">
      <span class="pcard-ovr-bubble" style="background:${getTeamColor(team)}">${ovr}</span>
    </div>
  ` : '';
  
  // Team header
  const teamHeader = `
    <div class="pcard-team-header" style="--team-primary:${getTeamColor(team)};">
      <img class="pcard-logo" src="${getTeamLogo(team)}" alt="${team}">
      <span class="pcard-team-name">${team}</span>
    </div>
  `;
  
  // Player info
  const playerInfo = `
    <div class="pcard-player">
      <span class="pcard-name ${ovr>=90?'shimmer':ovr>=85?'vibrant':''}">${name}</span>
      <span class="pcard-pos">${POS_DISPLAY[pos]||pos}</span>
      <span class="pcard-ovr ${ovr>=90?'shimmer':ovr>=85?'vibrant':''}">${ovr} OVR</span>
    </div>
  `;
  
  // Wins bar
  const w = Math.max(r.v, 0), gp = sim ? Math.max(0, Math.min(w, r.pri)) : 0, grey = w - gp;
  const mx = Math.max(0.3, ...entries.filter(r=>r.g===g).map(r=>Math.max(r.v||0, 0)));
  const winsBar = `
    <div class="pcard-wins">
      <div class="pcard-wins-bar">
        <i class="pcard-wins-fill" style="width:${gp/mx*100}%;background:var(--team-primary,var(--gold));"></i>
        <i class="pcard-wins-grey" style="width:${grey/mx*100}%;background:var(--sub);opacity:.55;"></i>
      </div>
      <span class="pcard-wins-val">${wavg(r.v)}</span>
    </div>
  `;
  
  // Footer with pick info
  const pickInfo = r.overall ? `
    <div class="pcard-footer">
      <span class="pcard-pick">R${r.round}.${String(r.pickInRound).padStart(2,'0')}</span>
      <span class="pcard-wins-small">(${r.v.toFixed(2)})</span>
    </div>
  ` : '';
  
  // Draft button
  const draftBtn = btn(r);
  
  return `
    <div class="pcard" style="--team-primary:${getTeamColor(team)};">
      <div class="pcard-header">${teamHeader}${ovrBubbles}</div>
      <div class="pcard-body">
        ${playerInfo}
        ${winsBar}
        ${pickInfo}
        <div class="pcard-action">${draftBtn}</div>
      </div>
    </div>
  `;
}
```

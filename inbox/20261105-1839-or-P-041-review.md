```js
let fadeTimers = {}; // track pending fade timers per view

function setActiveTab(viewId, tabBtnId) {
  const current = document.querySelector('.view.active');
  const next = document.getElementById(viewId);
  if (!current || !next || current === next) return;
  
  // Cancel any pending fade-out on current
  if (fadeTimers.current) { clearTimeout(fadeTimers.current); fadeTimers.current = null; }
  if (fadeTimers.next) { clearTimeout(fadeTimers.next); fadeTimers.next = null; }
  
  // Fade out current (2s)
  current.classList.add('fade-out');
  current.style.pointerEvents = 'none';
  fadeTimers.current = setTimeout(() => {
    current.classList.remove('active', 'fade-out');
    current.style.pointerEvents = '';
    
    next.classList.add('fade-in');
    next.classList.add('active');
    requestAnimationFrame(() => {
      next.classList.remove('fade-in');
    });
    fadeTimers.next = setTimeout(() => next.classList.remove('fade-in'), 2000);
  }, 2000);
  
  // Update tab button
  document.querySelectorAll('.tabbtn').forEach(b => b.classList.remove('active'));
  if (tabBtnId) document.getElementById(tabBtnId)?.classList.add('active');
}
```

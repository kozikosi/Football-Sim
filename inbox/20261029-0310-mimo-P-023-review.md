```
 function histBarsSVG(hist, maxH, userWins){
  const peak = Math.round(userWins), rowH = 22, W = 300, barMax = W - 96;
  let y = 0, rows = '';
  for(let w=0; w<=17; w++){
   const c = hist[w]||0, bw = maxH ? (c/maxH)*barMax : 0;
   rows += `<g transform="translate(0,${y})"><text x="34" y="14" text-anchor="end" font-size="11" fill="var(--sub)">${w}–${17-w}</text>`+
    `<rect x="42" y="3" width="${bw.toFixed(1)}" height="14" rx="2" fill="${w===peak?'var(--gold)':'var(--accent)'}"/>`+
    `<text x="${(42+bw+5).toFixed(1)}" y="14" font-size="11" fill="var(--text)">${c||''}</text></g>`;
   y += rowH;
  }
  return `<svg viewBox="0 0 ${W} ${y}" width="100%" height="${y}" role="img" aria-label="Seasons by win total">${rows}</svg>`;
 }
```

### JS Implementation (replace/modify `setActiveTab`)

```javascript
// P-050: Fade transition between views
const FADE_DURATION = 2000; // 2 seconds

async function setActiveTabWithFade(viewId, btnId) {
  // Skip fade if reduced motion
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    return setActiveTab(viewId, btnId);
  }

  const currentView = document.querySelector('.view-pane:not([hidden])');
  const targetView = document.getElementById(viewId);
  const hero = document.querySelector('.hero');
  
  if (!targetView || currentView === targetView) {
    return setActiveTab(viewId, btnId); // fallback to instant
  }

  // 1. Fade OUT current view + hero
  if (currentView) currentView.classList.add('fade-out');
  if (hero) hero.classList.add('fade-out');

  // Wait for fade out
  await new Promise(r => setTimeout(r, FADE_DURATION));

  // 2. Call original setActiveTab (handles hidden attrs, hero display, suppHead)
  setActiveTab(viewId, btnId);

  // 3. Fade IN new view + hero (if showing)
  const newView = document.getElementById(viewId);
  if (newView) newView.classList.add('fade-in');
  if (hero && !newView) { /* hero shown on Draft tab only */ hero.classList.add('fade-in'); }

  // 4. Cleanup classes after transition
  setTimeout(() => {
    if (currentView) currentView.classList.remove('fade-out');
    if (newView) newView.classList.remove('fade-in');
    if (hero) hero.classList.remove('fade-out', 'fade-in');
  }, FADE_DURATION + 100);
}
```

```javascript
function renderUnitEdges(homeTeam, awayTeam, g){
  const S = seasonState, H = S.unitRatings[homeTeam], A = S.unitRatings[awayTeam];
  const pl = g ? ensureGamePlans(g) : DEFAULT_PLAN_PAIR;
  const mh = matchupModel(H, A, lg, {off:pl.home.off, def:pl.away.def});
  const ma = matchupModel(A, H, lg, {off:pl.away.off, def:pl.home.def});
  
  const units = [
    {key:'olRun', label:'Run blocking', short:'run blocking'},
    {key:'olPass', label:'Pass protection', short:'pass protection'},
    {key:'rb', label:'Running backs', short:'running backs'},
    {key:'qb', label:'Quarterback', short:'quarterback play'},
    {key:'rec', label:'Receiving corps', short:'receiving corps'},
    {key:'passRush', label:'Pass rush', short:'pass rush'},
    {key:'frontRun', label:'Run defense', short:'run defense'},
    {key:'coverage', label:'Pass coverage', short:'secondary'},
    {key:'kicker', label:'Kicker', short:'kicker'}
  ];
  
  const rows = units.map(u=>{
    const av = A[u.key] || 0, hv = H[u.key] || 0;
    const diff = hv - av;
    const favorHome = diff > 0;
    const pct = Math.min(Math.abs(diff) / 30 * 100, 100); // cap at 30 pts = 100%
    
    return `
      <div class="unit-edge-row">
        <span class="unit-label">${u.label}</span>
        <div class="unit-bar-container">
          <div class="unit-bar ${favorHome ? 'favor-home' : 'favor-away'}" style="width:${pct}%"></div>
        </div>
        <span class="unit-value ${favorHome ? 'home' : 'away'}">${favorHome ? '+' : ''}${diff.toFixed(1)}</span>
      </div>
    `;
  }).join('');
  
  return `<div class="unit-edges">${rows}</div>`;
}
```

### 3. Victory Parallelogram Bar (Win Probability)

**Current:** Text like "The ratings lean KC by about 3.2 points (24.1–20.9)"

**New:** Visual parallelogram bar with team colors, numbers outside, no %.

```javascript
function renderVictoryBar(homeTeam, awayTeam, g){
  const S = seasonState, H = S.unitRatings[homeTeam], A = S.unitRatings[awayTeam];
  const pl = g ? ensureGamePlans(g) : DEFAULT_PLAN_PAIR;
  const mh = matchupModel(H, A, lg, {off:pl.home.off, def:pl.away.def});
  const ma = matchupModel(A, H, lg, {off:pl.away.off, def:pl.home.def});
  const projH = mh.structPts + HOME_FIELD, projA = ma.structPts;
  const spread = projH - projA;
  const homeWinProb = 1 / (1 + Math.exp(-spread / 4.5)); // approximate
  
  // Parallelogram: skew based on spread
  const skew = Math.min(Math.max(spread * 2, -30), 30); // degrees
  const leftPct = Math.round((1 - homeWinProb) * 100);
  const rightPct = Math.round(homeWinProb * 100);
  
  return `
    <div class="victory-bar" style="--skew:${skew}deg; --home-color:var(--team-primary, var(--gold)); --away-color:var(--opp-primary, var(--accent));">
      <div class="victory-track">
        <div class="victory-fill-away" style="width:${leftPct}%"></div>
        <div class="victory-fill-home" style="width:${rightPct}%"></div>
      </div>
      <div class="victory-labels">
        <span class="victory-score away">${Math.round(projA)}</span>
        <span class="victory-spread">${spread>=0?'+':''}${spread.toFixed(1)}</span>
        <span class="victory-score home">${Math.round(projH)}</span>
      </div>
      <div class="victory-teams">
        <span class="victory-team away" style="color:var(--opp-primary, var(--accent))">${nick(awayTeam)}</span>
        <span class="victory-team home" style="color:var(--team-primary, var(--gold))">${nick(homeTeam)}</span>
      </div>
    </div>
  `;
}
```

```javascript
function renderPointsVsAuto(homeTeam, awayTeam, g){
  const S = seasonState, H = S.unitRatings[homeTeam], A = S.unitRatings[awayTeam];
  const autoPlans = {home:{off:NEUTRAL_OFF, def:DEFAULT_DEF}, away:{off:NEUTRAL_OFF, def:DEFAULT_DEF}};
  const mhAuto = matchupModel(H, A, lg, {off:autoPlans.home.off, def:autoPlans.away.def});
  const maAuto = matchupModel(A, H, lg, {off:autoPlans.away.off, def:autoPlans.home.def});
  const projAutoH = mhAuto.structPts + HOME_FIELD;
  const projAutoA = maAuto.structPts;
  
  const pl = g ? ensureGamePlans(g) : DEFAULT_PLAN_PAIR;
  const mh = matchupModel(H, A, lg, {off:pl.home.off, def:pl.away.def});
  const ma = matchupModel(A, H, lg, {off:pl.away.off, def:pl.home.def});
  const projH = mh.structPts + HOME_FIELD;
  const projA = ma.structPts;
  
  const diffH = projH - projAutoH;
  const diffA = projA - projAutoA;
  
  return `
    <div class="pts-vs-auto">
      <span class="pts-label">Pts vs Auto</span>
      <span class="pts-diff ${diffA>=0?'pos':'neg'}">${nick(awayTeam)}: ${diffA>=0?'+':''}${diffA.toFixed(1)}</span>
      <span class="pts-diff ${diffH>=0?'pos':'neg'}">${nick(homeTeam)}: ${diffH>=0?'+':''}${diffH.toFixed(1)}</span>
    </div>
  `;
}
```

```javascript
function renderMatchupMathGraphical(homeTeam, awayTeam, g){

  const pl = g ? ensureGamePlans(g) : DEFAULT_PLAN_PAIR;

  const victoryBar = renderVictoryBar(homeTeam, awayTeam, g);
  const unitEdges = renderUnitEdges(homeTeam, awayTeam, g);
  const ptsVsAuto = renderPointsVsAuto(homeTeam, awayTeam, g);
  const planBubbles = renderPlanBubbles(g);
  
  return `
    <div class="matchup-math-graphical">

      <div class="math-sections">
        <section class="math-section">

          ${unitEdges}

        <section class="math-section">
          <h5>Plan</h5>

          ${ptsVsAuto}
        </section>
      </div>
    </div>
  `;
}
```

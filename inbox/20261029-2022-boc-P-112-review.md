### 1. `renderUserScheduleHTML()` (replace l.3857)

```javascript
function renderUserScheduleHTML(){
  const user = seasonUserTeam();
  const mine = seasonState.schedule.filter(g=>g.home===user||g.away===user).sort((a,b)=>a.week-b.week);
  const currentWeek = seasonState.currentWeek;
  const isPlayoffs = seasonState.phase === 'playoffs';
  const colorOf = (t) => (typeof getTeamColor === 'function') ? getTeamColor(t) : 'var(--gold)';

  const cards = mine.map(g=>{
    const home = g.home===user, opp = home ? g.away : g.home;
    const userColor = colorOf(user);
    const oppColor = colorOf(opp);
    const isCurrent = (g.week === currentWeek) && !seasonState.weekSimulated && !isPlayoffs;

    let scoreHTML = '', resultClass = '', cardAccent = '';
    if(g.played){
      const fs = home ? g.result.homeScore : g.result.awayScore;
      const os = home ? g.result.awayScore : g.result.homeScore;
      const won = fs > os;
      cardAccent = won ? userColor : oppColor;
      resultClass = won ? 'sched-win' : 'sched-loss';
      scoreHTML = `
        <div class="sched-score">
          <span class="sched-user-score" style="color:${userColor}">${fs}</span>
          <span class="sched-divider">–</span>
          <span class="sched-opp-score" style="color:${oppColor}">${os}</span>
          ${g.result.ot ? '<span class="sched-ot">OT</span>' : ''}
        </div>`;
    } else if(isCurrent){
      scoreHTML = `<div class="sched-upcoming"><span class="sched-time">Game Time</span></div>`;
      resultClass = 'sched-current';
    } else {
      scoreHTML = `<div class="sched-pending">—</div>`;
    }

    const played = !!g.played;
    const userLogo = (typeof zpbTeamLogo==='function') ? zpbTeamLogo(user) : (TEAM_ABBR[user]||user.slice(0,3).toUpperCase());
    const oppLogo  = (typeof zpbTeamLogo==='function') ? zpbTeamLogo(opp)  : (TEAM_ABBR[opp]||opp.slice(0,3).toUpperCase());
    const atvs = home ? 'vs' : '@';
    const clickAttr = played
      ? `tabindex="0" role="button" onclick="toggleRecap(${g.week})" onkeydown="if(event.key==='Enter'||event.key===' '){event.preventDefault();toggleRecap(${g.week});}"`
      : '';

    return `
      <div class="sched-card ${resultClass}" ${clickAttr} data-week="${g.week}" ${played?`style="--card-accent:${cardAccent};"`:''}>
        <div class="sched-week">WK ${g.week}</div>
        <div class="sched-matchup">
          <div class="sched-logo" style="color:${userColor}">${userLogo}</div>
          <div class="sched-atvs">${atvs}</div>
          <div class="sched-logo" style="color:${oppColor}">${oppLogo}</div>
        </div>
        <div class="sched-result">${scoreHTML}</div>
        ${g.divisional ? '<div class="sched-div-tag">DIV</div>' : ''}
      </div>`;
  }).join('');

  // recap details in a separate block container (not flex items)
  const recaps = mine
    .filter(g=>g.played && g.summary)
    .map(g=>`<div class="sched-recap-detail" id="recap-${g.week}" style="display:none;">
      ${renderGameResultHTML(g, user)}
    </div>`).join('');

  return `
    <div class="panel sched-panel" style="margin-bottom:14px;">
      <h4 style="margin-top:0;display:flex;align-items:center;gap:10px;">
        Your Schedule &amp; Results
        <span class="narr" style="font-weight:400;">(click a played game for recap)</span>
      </h4>
      <div class="sched-scroller"><div class="sched-cards">${cards}</div></div>
      <div class="sched-recap-list">${recaps}</div>
    </div>`;
}

function toggleRecap(week){
  const detail = document.getElementById('recap-'+week);
  if(detail) detail.style.display = detail.style.display === 'none' ? 'block' : 'none';
}
```

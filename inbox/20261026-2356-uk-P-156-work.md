```javascript
function renderStrengthsWeaknessesGraphic(user, S){
  const R = S.unitRatings[user], U = S.unitRanks[user], lg = S.league;
  const units = Object.keys(UNIT_INFO).filter(u=>u!=='kicker');
  
  const strengths = Object.keys(UNIT_INFO).filter(u=>u!=='kicker')
    .map(u=>({unit:u, label:UNIT_INFO[u].label, diff:R[u]-lg[u], rank:U[u]}))
    .filter(u=>u.diff > 1).sort((a,b)=>b.diff-a.diff).slice(0,3);
    
  const weaknesses = Object.keys(UNIT_INFO).filter(u=>u!=='kicker')
    .map(u=>({unit:u, label:UNIT_INFO[u].label, diff:R[u]-lg[u], rank:U[u]}))
    .filter(u=>u.diff < -1).sort((a,b)=>a.diff-b.diff).slice(0,3);
  
  return `
    <div class="sw-graphic">
      <div class="sw-section">
        <h4 class="sw-title foil-shimmer gold">STRENGTHS</h4>
        <div class="sw-list">
          ${strengths.map((s,i)=>`<div class="sw-item"><span class="sw-rank">#${i+1}</span><span class="sw-pos">${UNIT_INFO[s.unit].label.toUpperCase()}</span><span class="sw-diff pos">+${s.diff.toFixed(1)}</span></div>`).join('')}
        </div>
      </div>
      <div class="sw-section">
        <h4 class="sw-title">WEAKNESSES</h4>
        <div class="sw-list">
          ${weaknesses.map((s,i)=>`<div class="sw-item"><span class="sw-rank">#${i+1}</span><span class="sw-pos">${UNIT_INFO[s.unit].label.toUpperCase()}</span><span class="sw-diff neg">${s.diff.toFixed(1)}</span></div>`).join('')}
        </div>
      </div>
    </div>
  `;
}
```

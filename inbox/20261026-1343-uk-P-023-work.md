### 2. Histogram Chart (Aligned)
**Current**: Flex bars with percentage widths - misaligned labels
**Proposed**: SVG horizontal bar chart with proper alignment

```javascript
function renderHistogramSVG(hist, maxH, userWins) {
  const maxWin = Object.keys(hist).reduce((a,k)=>Math.max(a,parseInt(k)),0);
  const minWin = Object.keys(hist).reduce((a,k)=>Math.min(a,parseInt(k)),Infinity);
  const barHeight = 20;
  const chartHeight = (maxWin - minWin + 1) * (barHeight + 4);
  const maxBarWidth = 280;
  
  let svg = `<svg width="100%" height="${chartHeight}" viewBox="0 0 ${maxBarWidth + 60} ${chartHeight}">`;
  for(let w = minWin; w <= maxWin; w++) {
    const count = hist[w] || 0;
    const pct = maxH ? count / maxH : 0;
    const barWidth = pct * maxBarWidth;
    const isUserPeak = w === Math.round(userWins);
    svg += `
      <g transform="translate(50, ${(w-minWin)*(barHeight+4)})">
        <text x="-8" y="${barHeight/2+4}" text-anchor="end" font-size="10" fill="var(--sub)">${w}</text>
        <rect x="0" y="0" width="${barWidth}" height="${barHeight}" 
              fill="${isUserPeak ? 'var(--gold)' : 'var(--accent)'}" rx="2"/>
        ${count ? `<text x="${barWidth+4}" y="${barHeight/2+4}" font-size="10" fill="var(--text)">${count}</text>` : ''}
      </g>`;
  }
  svg += `</svg>`;
  return svg;
}
```

## Proposed Implementation

### 1. CSS Fix (in `<style>` block)

```css
/* P-055: Pick Priority menu - no green in light mode */
#draftSort2 option[value="priority"],
#draftSort2 option[value="value"] {
  /* Remove any hardcoded green */
  color: inherit;
}

/* Ensure select uses team palette when team selected */
#draftSort2 {
  color: var(--text);
  background: var(--panel2);
  border-color: var(--border);
}

/* When team selected, style the priority/value options with team colors */
[data-team] #draftSort2 option[value="priority"] {
  color: var(--team-primary);
}
[data-team] #draftSort2 option[value="value"] {
  color: var(--team-secondary);
}

/* Fallback: use accent colors when no team */
:root:not([data-team]) #draftSort2 option[value="priority"] {
  color: var(--accent);
}
:root:not([data-team]) #draftSort2 option[value="value"] {
  color: var(--accent2);
}
```

**Note:** `<option>` styling is limited by browser. Better approach: style the select itself and use custom dropdown (like the position filter `.posdrop`/`.posbtn`).

### 3. JS Integration

```javascript
// Initialize draft sort dropdown (similar to position filter)
function initDraftSortDropdown() {
  const btn = document.getElementById('draftSortBtn');
  const list = document.getElementById('draftSortList');
  const label = document.getElementById('draftSortLabel');
  
  if (!btn || !list) return;
  
  // Toggle
  btn.addEventListener('click', () => {
    const expanded = btn.getAttribute('aria-expanded') === 'true';
    btn.setAttribute('aria-expanded', !expanded);
    list.hidden = expanded;
  });
  
  // Option selection
  list.querySelectorAll('.posopt').forEach(opt => {
    opt.addEventListener('click', () => {
      const value = opt.dataset.value;
      const text = opt.textContent;
      
      // Update button label
      label.textContent = text;
      
      // Update aria-selected
      list.querySelectorAll('.posopt').forEach(o => o.setAttribute('aria-selected', 'false'));
      opt.setAttribute('aria-selected', 'true');
      
      // Close dropdown
      btn.setAttribute('aria-expanded', 'false');
      list.hidden = true;
      
      // Trigger re-render (existing logic)
      const sortSelect = document.getElementById('draftSort2');
      if (sortSelect) {
        sortSelect.value = value;
        sortSelect.dispatchEvent(new Event('change'));
      } else {
        // Fallback: call rer() directly if sortSelect removed
        rer();
      }
    });
  });
  
  // Close on outside click
  document.addEventListener('click', (e) => {
    if (!btn.contains(e.target) && !list.contains(e.target)) {
      btn.setAttribute('aria-expanded', 'false');
      list.hidden = true;
    }
  });
}

// Initialize on load
document.addEventListener('DOMContentLoaded', initDraftSortDropdown);

// Update option colors when team changes
function updateDraftSortColors() {
  const hasTeam = document.documentElement.hasAttribute('data-team');
  const list = document.getElementById('draftSortList');
  if (!list) return;
  
  list.querySelectorAll('.posopt').forEach(opt => {
    if (opt.dataset.value === 'priority') {
      opt.style.color = hasTeam ? 'var(--team-primary)' : 'var(--accent)';
    } else if (opt.dataset.value === 'value') {
      opt.style.color = hasTeam ? 'var(--team-secondary)' : 'var(--accent2)';
    } else {
      opt.style.color = 'var(--text)';
    }
  });
}

// Call in applyTeamColorsToUI()
function applyTeamColorsToUI(teamName) {
  // ... existing code ...
  updateDraftSortColors();
}
```

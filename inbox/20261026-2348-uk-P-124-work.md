```javascript
function renderSeasonReviewVisual(F){
  const {user, me, N, S, log, st} = F;
  const isChampion = S.champion === user;
  const U = S.unitRanks[user], R = S.unitRatings[user];
  const lg = S.league;
  
  // Compute unit strengths/weaknesses
  const units = Object.keys(UNIT_INFO).filter(u=>u!=='kicker');
  const unitData = units.map(u=>{
    const rank = U[u], rating = R[u], lgAvg = lg[u];
    const diff = rating - lgAvg;
    const val = unitValuePerPoint(u);
    const winGain = Math.max(0, (lgAvg - rating) * val * 17 / POINTS_PER_WIN);
    return {unit: u, label: UNIT_INFO[u].label, rank, rating, lgAvg, diff, val, winGain};
  });
  
  // Sort by win gain (weaknesses first for non-champions)
  const weaknesses = unitData.filter(u=>u.diff < -1).sort((a,b)=>a.diff-b.diff).slice(0,4);
  const strengths = unitData.filter(u=>u.diff > 1).sort((a,b)=>b.diff-a.diff).slice(0,4);
  
  // Champion: only strengths, more of them
  const displayStrengths = isChampion ? unitData.filter(u=>u.diff > 0).sort((a,b)=>b.diff-a.diff).slice(0,6) : strengths;
  const displayWeaknesses = isChampion ? [] : weaknesses;
  
  // Strength/Weakness Cards
  function renderUnitCard(u, type){
    const isStrong = type === 'strength';
    const player = getTopPlayerInUnit(u.unit);
    const color = u.diff > 0 ? 'var(--gold)' : 'var(--bad)';
    const icon = isStrong ? '🏆' : '⚠️';
    const label = isStrong ? 'STRENGTH' : 'WEAKNESS';
    
    return `
      <div class="sr-unit-card ${u.diff>0?'strength':'weakness'}" style="--unit-color:${color};">
        <div class="sru-header">
          <span class="sru-icon">${isStrong?'🏆':'⚠️'}</span>
          <span class="sru-label">${label}</span>
          <span class="sru-unit">${UNIT_INFO[u.unit].label.toUpperCase()}</span>
        </div>
        <div class="sru-diff ${u.diff>0?'pos':'neg'}">${u.diff>=0?'+':''}${u.diff.toFixed(1)} vs lg avg</div>
        <div class="sru-win">${u.winGain.toFixed(1)} wins on table</div>
        ${player ? `
          <div class="sru-player">
            <span class="sru-player-name ${player.ovr>=90?'shimmer':player.ovr>=85?'vibrant':''}">${player.name}</span>
            <span class="sru-player-meta">${LABELS[player.group]} · ${player.ovr} OVR</span>
            <span class="sru-player-impact">${fmtWar(getPlayerUnitImpact(player))} wins</span>
          </div>
        ` : ''}
      </div>
    `;
  }
  
  // Champion badge
  const championBadge = isChampion ? `
    <div class="sr-champion-banner foil-shimmer rose">
      🏆 ${N} ARE SUPER BOWL CHAMPIONS!
    </div>
  ` : '';
  
  // Try again button
  const tryAgainBtn = `
    <div class="sr-tryagain">
      <button class="resetbtn zpb-tryagain" id="tryAgain20Btn">
        <span class="tryagain-text">TRY AGAIN TO </span>
        <span class="tryagain-20 foil-shimmer gold">20-0</span>
        <span class="tryagain-q foil-shimmer gold">?</span>
      </button>
    </div>
  `;
  
  return `
    <div class="season-review-visual">
      <header class="sr-header">
        <div class="sr-team" style="--team-primary:${getTeamColor(user)};">
          <div class="sr-logo">${typeof zpbTeamLogo==='function'?zpbTeamLogo(user):TEAM_ABBR[user]}</div>
          <div class="sr-name">${user}</div>
          ${isChampion ? '<span class="sr-champion-tag foil-shimmer rose">CHAMPIONS</span>' : ''}
        </div>
        <div class="sr-record">${F.me.wins}-${F.me.losses}${F.me.ties?'-'+F.me.ties:''}</div>
      </header>
      
      ${isChampion ? `
        <div class="sr-champion-section">
          <h3 class="sr-section-title foil-shimmer gold">🏆 CHAMPIONS — ONLY STRENGTHS</h3>
          <div class="sr-strengths-grid">
            ${displayStrengths.map(u=>renderUnitCard(u, 'strength')).join('')}
          </div>
        </div>
      ` : `
        <div class="sr-sw-section">
          <div class="sr-strengths">
            <h3 class="sr-section-title">🏆 STRENGTHS</h3>
            <div class="sr-cards-grid">
              ${strengths.map(u=>renderUnitCard(u, 'strength')).join('')}
            </div>
          </div>
          <div class="sr-weaknesses">
            <h3 class="sr-section-title">⚠️ WEAKNESSES</h3>
            <div class="sr-cards-grid">
              ${weaknesses.map(u=>renderUnitCard(u, 'weakness')).join('')}
            </div>
          </div>
        </div>
      `}
      
      <div class="sr-tryagain-section">
        ${tryAgainBtn}
      </div>
    </div>
  `;
}
```

### 2. Helper: Get Top Player in Unit

```javascript
function getTopPlayerInUnit(unit){
  const mine = teamPlayerStats(seasonUserTeam());
  const slots = UNIT_INFO[unit]?.slots || [];
  const players = mine.filter(p=>slots.includes(p.slot));
  if(!players.length) return null;
  return players.sort((a,b)=>((b.ovr||0)-(a.ovr||0)))[0];
}

function getPlayerUnitImpact(player){
  // Use existing win value calculation
  return playerWar(player).war;
}
```

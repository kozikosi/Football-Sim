## CURRENT NFL OVERTIME RULES (2025 SEASON)

### Regular Season (NEW as of 2025)
- **Period length:** 10 minutes (changed from 15 in 2017)
- **Both teams guaranteed a possession** — even if first possession scores a TD
- **If Team A scores TD on first possession:** Team B gets a possession to match or win
- **If Team A scores FG on first possession:** Team B gets a possession; if Team B scores TD → Team B wins; if Team B scores FG → sudden death continues
- **If neither scores on first possession:** Sudden death (next score wins)
- **Tie possible:** If both teams match scores and time expires, or if neither scores in 10 minutes
- **Coin toss:** Winner chooses to receive or defer (same as before)

### Postseason (since 2022)
- **Period length:** 15 minutes
- **Both teams guaranteed a possession** — even if first possession scores TD
- **No ties** — continue periods until winner
- **Same possession rules as new regular season**

### Historical Context
| Era | Regular Season | Postseason | Notes |
|-----|---------------|------------|-------|
| 1974-2011 | 15 min sudden death | 15 min sudden death | First score wins |
| 2012-2016 | 15 min modified sudden death | 15 min modified sudden death | FG on first possession doesn't end game; TD does |
| 2017-2021 | 10 min modified sudden death | 15 min modified sudden death | Regular season shortened to 10 min |
| 2022-2024 | 10 min modified sudden death | 15 min both teams possess | Postseason changed after 2021 Chiefs-Bills |
| **2025+** | **10 min both teams possess** | **15 min both teams possess** | **Rules now aligned** |

```js
function simulateOvertime(game, isPlayoff = false) {
  const periodLength = isPlayoff ? 15 : 10; // minutes
  const bothPossess = true; // NEW: both teams guaranteed possession
  
  let teamA = game.home, teamB = game.away;
  let tossWinner = Math.random() < 0.5 ? teamA : teamB;
  let firstPossession = tossWinner;
  let secondPossession = firstPossession === teamA ? teamB : teamA;
  
  let scoreA = game.result.homeScore;
  let scoreB = game.result.awayScore;
  let timeRemaining = periodLength * 60; // seconds
  let possessions = 0;
  
  while (timeRemaining > 0) {
    if (possessions === 0) {
      // First possession
      const result = simulateDrive(firstPossession, secondPossession, /* start */ 25, 'OT');
      scoreA += firstPossession === teamA ? result.points : 0;
      scoreB += firstPossession === teamB ? result.points : 0;
      timeRemaining -= result.time;
      possessions++;
      
      if (result.outcome === 'TD' && bothPossess) {
        // Second team gets chance to match
        continue; // loop to second possession
      }
      if (result.outcome === 'FG' && bothPossess) {
        // Second team gets chance; if they score TD they win, FG ties and continues
        continue;
      }
      if (result.outcome === 'TO' || result.outcome === 'PUNT') {
        // Sudden death begins
        bothPossess = false;
        continue;
      }
    } else if (possessions === 1) {
      // Second possession (guaranteed under new rules)
      const result = simulateDrive(secondPossession, firstPossession, /* start */ 25, 'OT');
      scoreA += secondPossession === teamA ? result.points : 0;
      scoreB += secondPossession === teamB ? result.points : 0;
      timeRemaining -= result.time;
      possessions++;
      
      // Check win conditions
      if (scoreA !== scoreB) {
        // Leader wins (unless both had TDs? - rules say next score wins after both possess)
        break;
      }
      // If tied after both possessions, sudden death
      bothPossess = false;
    } else {
      // Sudden death: next score wins
      const result = simulateDrive(possessions % 2 === 0 ? firstPossession : secondPossession, 
                                    possessions % 2 === 0 ? secondPossession : firstPossession, 
                                    25, 'OT');
      scoreA += (possessions % 2 === 0 ? firstPossession : secondPossession) === teamA ? result.points : 0;
      scoreB += (possessions % 2 === 0 ? firstPossession : secondPossession) === teamB ? result.points : 0;
      timeRemaining -= result.time;
      if (result.points > 0) break;
      possessions++;
    }
  }
  
  // Tie if still tied and time expired (regular season only)
  if (!isPlayoff && scoreA === scoreB) {
    return { outcome: 'TIE', homeScore: scoreA, awayScore: scoreB };
  }
  
  return { outcome: scoreA > scoreB ? 'HOME' : 'AWAY', homeScore: scoreA, awayScore: scoreB };
}
```

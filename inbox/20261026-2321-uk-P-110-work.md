### 4. Matchup Math in Roster Order (Depth Chart Order)

**New function: `renderLineupMatchupMath(roster, hostId)`**

```javascript
function renderLineupMatchupMath(roster, hostId){
  const S = seasonState;
  if(!S || !S.opponent) return ''; // Only show when opponent known
  
  const opp = S.opponent;
  const oppRatings = S.unitRatings[opp];
  const myRatings = S.unitRatings[seasonUserTeam()];
  const oppRanks = S.unitRanks[opp];
  const myRanks = S.unitRanks[seasonUserTeam()];
  
  // Map depth chart slots to units
  const slotToUnit = {
    'DE1':'passRush','DE2':'passRush','DT1':'frontRun','DT2':'frontRun',
    'MIKE':'frontRun','WILL':'frontRun','SAM':'frontRun',
    'CB1':'coverage','CB2':'coverage','CB3':'coverage','SS':'coverage','FS':'coverage',
    'QB':'qb','RB1':'rb','RB2':'rb','WR1':'rec','WR2':'rec','WR3':'rec','TE1':'rec','TE2':'rec',
    'OT1':'olPass','OT2':'olPass','OG1':'olRun','OG2':'olRun','C':'olRun',
    'K':'kicker'
  };
  
  const rows = ROSTER_SLOTS.map(sl=>{
    const p = roster.slots[sl];
    if(!p) return '';
    const unit = slotToUnit[sl];
    if(!unit) return '';
    
    const myVal = myRatings[unit] || 0;
    const oppVal = oppRatings[unit] || 0;
    const myRank = myRanks[unit] || 32;
    const oppRank = oppRanks[unit] || 32;
    const diff = myVal - oppVal;
    const favor = diff > 0 ? 'us' : diff < 0 ? 'them' : 'even';
    
    const star90 = p[3] >= 90;
    const star85 = p[3] >= 85;
    const nameStyle = star90 ? 'shimmer' : star85 ? 'vibrant' : '';
    
    return `
      <div class="matchup-row ${favor}" data-slot="${sl}">
        <span class="matchup-slot">${LABELS[sl]||sl}</span>
        <span class="matchup-player ${nameStyle}">${p[0]}</span>
        <span class="matchup-unit">${UNIT_INFO[unit]?.label || unit}</span>
        <div class="matchup-bar">
          <div class="matchup-fill ${favor}" style="width:${Math.min(Math.abs(diff)/15*100, 100)}%"></div>
        </div>
        <span class="matchup-diff ${favor}">${diff>=0?'+':''}${diff.toFixed(1)}</span>
        <span class="matchup-rank">#${myRank} vs #${oppRank}</span>
      </div>
    `;
  }).filter(Boolean).join('');
  
  if(!rows) return '';
  
  return `
    <details class="panel" style="margin-top:12px;">
      <summary style="cursor:pointer;font-weight:700;">Matchup Math (roster order)</summary>
      <div class="matchup-math-roster">${rows}</div>
    </details>
  `;
}
```

### 5. Group Scores → Team-Colored Cards

**Modify `lineupScoresHTML` (line 1256):**

```javascript
function lineupScoresHTML(roster, hostId){
  const mw = modelWeights(), user = seasonUserTeam();
  const teamColor = getTeamColor(user);
  const cells = LINEUP_GROUPS.map(([name, slots])=>{
    let sum = 0, wt = 0; 
    slots.forEach(sl=>{ const p = roster.slots[sl]; const w = mw[sl]||0; sum += (p ? effOvr(p, sl) : 70)*w; wt += w; });
    const v = wt ? sum/wt : 0, prev = _lineupPrev[hostId+name], d = prev==null ? 0 : v-prev;
    _lineupPrev[hostId+name] = v;
    const arrow = d>=0.05 ? '▲' : d<=-0.05 ? '▼' : '';
    const arrowColor = d>0 ? teamColor : d<0 ? 'var(--bad)' : '';
    return `<div style="flex:1;min-width:120px;text-align:center;border-left:3px solid ${teamColor};">
      <div style="font-size:1.3rem;font-weight:800;color:${teamColor};">${v.toFixed(1)}${arrow?` <span style="font-size:0.75rem;color:${arrowColor};">${arrow}${Math.abs(d).toFixed(1)}</span>`:''}</div>
      <div class="narr">${name}</div>
    </div>`;
  }).join('');
  return `<div class="narr" style="margin:6px 0 2px;"><b>Group scores</b> (weighted by win value; flex penalties included)</div>
    <div style="display:flex;gap:6px;flex-wrap:wrap;margin-bottom:8px;">${cells}</div>`;
}
```

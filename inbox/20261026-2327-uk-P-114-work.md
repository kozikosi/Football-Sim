### 4. Sim Rest of Season → Summary Panel (NEW)

**Current flow:** `simRestOfSeason()` → `regularReview`

**New flow:** `simRestOfSeason()` → `renderSeasonSummaryPanel()` → user clicks "Continue to Review" → `regularReview`

**New function `renderSeasonSummaryPanel(container)`:**

```javascript
function renderSeasonSummaryPanel(container){
  const user = seasonUserTeam();
  const st = computeStandings(seasonState.schedule, teamNames());
  const s = st[user];
  const conf = conferenceOfTeam(user);
  const seed = seasonState.seeds[conf].find(x=>x.team===user);
  const isOneSeed = seed && seed.seed === 1;
  const madePlayoffs = seed !== undefined;
  const champ = seasonState.champion;
  const wonSuperBowl = champ === user;
  
  // Compute key stats
  const pf = s.pf, pa = s.pa, pd = pf - pa;
  const wins = s.wins, losses = s.losses, ties = s.ties;
  const record = `${wins}-${losses}${ties?'-'+ties:''}`;
  const divRank = divisionStandings(conferenceOfTeam(user), computeStandings(seasonState.schedule, teamNames()), seasonState.schedule).findIndex(d=>d.team===user) + 1;
  
  // Awards summary
  const awards = seasonState.awardsRS;
  const userAwards = awards.individual.filter(a=>(a.c && a.c.team===user) || a.team===user);
  const mvp = userAwards.find(a=>a.title.startsWith('Most'));
  const opoy = userAwards.find(a=>a.title.startsWith('Offensive'));
  const dpoy = userAwards.find(a=>a.title.startsWith('Defensive'));
  const coach = userAwards.find(a=>a.title.startsWith('Coach'));
  
  // Badges for summary
  const badges = [];
  badges.push(`<span class="summary-badge record">${record}</span>`);
  badges.push(`<span class="summary-badge pf">${pf} PF</span>`);
  badges.push(`<span class="summary-badge pa">${pa} PA</span>`);
  badges.push(`<span class="summary-badge pd ${pd>=0?'pos':'neg'}">${pd>=0?'+':''}${pd} PD</span>`);
  if(seed) badges.push(`<span class="summary-badge seed">#${seed.seed} Seed</span>`);
  badges.push(`<span class="summary-badge div">Div #${divRank}</span>`);
  if(madePlayoffs) badges.push(`<span class="summary-badge playoffs">PLAYOFFS</span>`);
  if(isOneSeed) badges.push(`<span class="summary-badge one-seed shimmer">#1 SEED</span>`);
  if(wonSuperBowl) badges.push(`<span class="summary-badge champ shimmer">CHAMPIONS</span>`);
  
  // Key players
  const topPlayers = Object.values(seasonState.playerStats[user]||{}).sort((a,b)=>(b.st.passYds||0)+(b.st.rushYds||0)*1.4+(b.st.recYds||0)+(b.st.sack||0)*40+(b.st.tkl||0)*4 - ((a.st.passYds||0)+(a.st.rushYds||0)*1.4+(a.st.recYds||0)+(a.st.sack||0)*40+(a.st.tkl||0)*4)).slice(0,3);
  
  const playerCards = topPlayers.map(p=>`
    <div class="summary-player-card">
      <div class="spc-avatar" style="background:${getTeamColor(user)}">${p.name.charAt(0)}</div>
      <div class="spc-info">
        <div class="spc-name ${p.ovr>=90?'shimmer':''}">${p.name}</div>
        <div class="spc-meta">${LABELS[p.group]||p.group} · ${p.ovr} OVR</div>
        <div class="spc-stats">${formatPlayerStats(p)}</div>
      </div>
    </div>
  `).join('');
  
  // Visual elements
  const confetti = wonSuperBowl || isOneSeed ? '<div class="summary-confetti"></div>' : '';
  const champBanner = wonSuperBowl ? `<div class="summary-champ-banner"><span class="foil-shimmer rose">CHAMPIONS</span></div>` : '';
  const oneSeedBanner = isOneSeed ? `<div class="summary-one-seed-banner"><span class="foil-shimmer gold">#1 SEED</span></div>` : '';
  
  return `
    <div class="panel summary-panel">
      ${confetti}
      ${champBanner}
      ${oneSeedBanner}
      
      <div class="summary-header">
        <div class="summary-team" style="--team-primary:${getTeamColor(seasonUserTeam())};">
          <div class="summary-logo">${typeof zpbTeamLogo==='function'?zpbTeamLogo(seasonUserTeam()):TEAM_ABBR[seasonUserTeam()]}</div>
          <div class="summary-name">${seasonUserTeam()}</div>
        </div>
        <div class="summary-record">${computeStandings(seasonState.schedule, teamNames())[seasonUserTeam()].wins}-${st.losses}${st.ties?'-'+st.ties:''}</div>
      </div>
      
      <div class="summary-badges">${badges.join('')}</div>
      
      <div class="summary-stats">
        <div class="summary-stat"><span class="summary-stat-val">${pf}</span><span class="summary-stat-lbl">POINTS FOR</span></div>
        <div class="summary-stat"><span class="summary-stat-val">${pa}</span><span class="summary-stat-lbl">POINTS AGAINST</span></div>
        <div class="summary-stat"><span class="summary-stat-val ${pd>=0?'pos':'neg'}">${pd>=0?'+':''}${pd}</span><span class="summary-stat-lbl">POINT DIFF</span></div>
        <div class="summary-stat"><span class="summary-stat-val">${mvp?'MVP':''}</span><span class="summary-stat-lbl">MVP</span></div>
      </div>
      
      <div class="summary-players">
        <h4>SEASON STARS</h4>
        <div class="summary-player-row">${playerCards}</div>
      </div>
      
      <div class="summary-actions">
        <button class="resetbtn zpb-continue" id="continueToReviewBtn">Continue to Review</button>
      </div>
    </div>
  `;
}
```

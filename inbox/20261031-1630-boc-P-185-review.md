// Fix 1: reconcile algorithm with table — use contrast ratio, not just L*
function toneLift(hex, targetContrast = 4.5) {
  // Simplified: if contrast on white < targetContrast, lift lightness until it passes
  // Use HSL for simplicity (no OKLCH helpers needed)
  const hsl = hexToHsl(hex);
  let L = hsl.L;
  while (contrastOnWhite(hslToHex(hsl.H, hsl.S, L)) < targetContrast && L < 0.85) {
    L += 0.02;
  }
  return hslToHex(hsl.H, hsl.S, L);
}

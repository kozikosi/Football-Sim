function updateNextPickDisplay() {
  const pos = draftSim.pos; // 0-indexed
  const round = Math.floor(pos / 32) + 1;
  const pick = pos + 1;
  const display = document.getElementById('nextPickDisplay');
  if (display) {
    display.querySelector('.pick-badge').textContent = `(${round}.${pick})`;

}

Since native `<select>` cannot style individual options with colors, we need a custom dropdown component that:
1. Uses a button + hidden list (accessible pattern)
2. Shows team abbreviation + colored indicator

### 4. JavaScript Implementation

Replace the team select population in `initDraftSetup()`:

```javascript
function initDraftSetup() {
  // ... existing code ...

  // Team dropdown (custom)
  const teamSelect = document.getElementById('draftTeamSelect');
  const trigger = teamSelect.querySelector('.custom-select-trigger');
  const optionsList = teamSelect.querySelector('.custom-select-options');
  const teamNames = Object.keys(TEAM_DATA).sort();
  
  // Populate options
  teamNames.forEach((team, index) => {
    const colors = TEAM_COLORS[team] || { primary: '#666', secondary: '#999', accent: '#ccc' };
    const abbr = TEAM_ABBR[team] || team.substring(0,3).toUpperCase();
    const li = document.createElement('li');
    li.className = 'custom-select-option';
    li.role = 'option';
    li.tabIndex = -1;
    li.dataset.team = team;
    li.innerHTML = `
      <span class="team-color-dot" style="background:${colors.primary}"></span>
      <span class="team-abbr">${abbr}</span>
      <span class="team-full">${team}</span>
    `;
    optionsList.appendChild(li);
  });
  
  // Random option
  const randomLi = document.createElement('li');
  randomLi.className = 'custom-select-option is-random';
  randomLi.role = 'option';
  randomLi.tabIndex = -1;
  randomLi.dataset.team = '__random';
  randomLi.innerHTML = `
    <span class="team-color-dot" style="background:linear-gradient(135deg,var(--gold),var(--accent2))"></span>
    <span class="team-abbr">RND</span>
    <span class="team-full">Random team</span>
  `;
  optionsList.insertBefore(randomLi, optionsList.firstChild);
  
  // Set default (Chiefs or first)
  const defaultTeam = teamNames.includes('Kansas City Chiefs') ? 'Kansas City Chiefs' : teamNames[0];
  selectTeam(defaultTeam);
  
  // Trigger click
  trigger.addEventListener('click', (e) => {
    e.stopPropagation();
    toggleDropdown();
  });
  
  // Keyboard navigation
  trigger.addEventListener('keydown', (e) => {
    if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
      e.preventDefault();
      openDropdown();
      const options = optionsList.querySelectorAll('.custom-select-option:not([hidden])');
      const first = options[0];
      if (first) first.focus();
    } else if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      toggleDropdown();
    } else if (e.key === 'Escape') {
      closeDropdown();
    }
  });
  
  // Options keyboard
  optionsList.addEventListener('keydown', (e) => {
    const options = Array.from(optionsList.querySelectorAll('.custom-select-option:not([hidden])'));
    const currentIndex = options.findIndex(opt => opt === document.activeElement);
    if (e.key === 'ArrowDown') {
      e.preventDefault();
      const next = options[(currentIndex + 1) % options.length];
      next?.focus();
    } else if (e.key === 'ArrowUp') {
      e.preventDefault();
      const prev = options[(currentIndex - 1 + options.length) % options.length];
      prev?.focus();
    } else if (e.key === 'Enter' || e.key === ' ') {
      e.preventDefault();
      selectOption(document.activeElement);
      closeDropdown();
    } else if (e.key === 'Escape') {
      closeDropdown();
      trigger.focus();
    } else if (e.key === 'Tab') {
      closeDropdown();
    }
  });
  
  // Click outside to close
  document.addEventListener('click', (e) => {
    if (!teamSelect.contains(e.target)) closeDropdown();
  });
  
  // Option click
  optionsList.addEventListener('click', (e) => {
    const option = e.target.closest('.custom-select-option');
    if (option) {
      selectOption(option);
      closeDropdown();
    }
  });
  
  function toggleDropdown() {
    const expanded = teamSelect.getAttribute('aria-expanded') === 'true';
    if (expanded) closeDropdown(); else openDropdown();
  }
  
  function openDropdown() {
    teamSelect.setAttribute('aria-expanded', 'true');
    optionsList.hidden = false;
    // Position check for viewport
    const rect = optionsList.getBoundingClientRect();
    if (rect.bottom > window.innerHeight - 20) {
      optionsList.style.maxHeight = `${window.innerHeight - rect.top - 40}px`;
    }
  }
  
  function closeDropdown() {
    teamSelect.setAttribute('aria-expanded', 'false');
    optionsList.hidden = true;
  }
  
  function selectOption(option) {
    const team = option.dataset.team;
    selectTeam(team);
    trigger.focus();
  }
  
  function selectTeam(team) {
    // Update trigger display
    const colors = team === '__random' ? { primary: 'linear-gradient(135deg,var(--gold),var(--accent2))', secondary: '', accent: '' } : (TEAM_COLORS[team] || { primary: '#666', secondary: '#999', accent: '#ccc' });
    const abbr = team === '__random' ? 'RND' : (TEAM_ABBR[team] || team.substring(0,3).toUpperCase());
    const name = team === '__random' ? 'Random team' : team;
    
    trigger.querySelector('.team-indicator').style.background = colors.primary;
    trigger.querySelector('.team-name').textContent = name;
    
    // Update aria-selected
    optionsList.querySelectorAll('.custom-select-option').forEach(opt => {
      opt.setAttribute('aria-selected', opt.dataset.team === team);
    });
    
    // Update hidden native select for form compatibility
    const nativeSel = document.getElementById('draftSlotSelect');
    if (nativeSel) nativeSel.value = team;
    
    // Apply team colors if team selected (not random)
    if (team !== '__random' && window.applyTeamColors) {
      applyTeamColors(team);
    } else if (team === '__random' && window.applyTeamColors) {
      // Reset to default rose/gold
      document.documentElement.removeAttribute('data-team');
      document.documentElement.style.removeProperty('--tp');
      document.documentElement.style.removeProperty('--ts');
      document.documentElement.style.removeProperty('--ta');
    }
  }
  
  // Pick number dropdown (native, no logo)
  const pickSel = document.getElementById('draftPickNumberSelect');
  pickSel.innerHTML = '<option value="random">Random</option>';
  for(let i=1;i<=32;i++){
    const o=document.createElement('option');
    o.value=i;
    o.textContent='Pick '+i;
    pickSel.appendChild(o);
  }
  pickSel.value = 'random';
  
  // ... rest of initDraftSetup (event listeners)
}
```

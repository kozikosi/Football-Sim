### 1. New `renderAwardCardsHTML(awards, type)` Function

```javascript
function renderAwardCardsHTML(awards, type){
  const user = seasonUserTeam();
  const isPlayoff = type === 'playoff';
  const A = isPlayoff ? awards : awards;
  
  let html = `<div class="awards-panel ${isPlayoff?'playoff':''}">`;
  
  // Individual Awards (MVP, OPOY, DPOY, Coach, SB MVP, etc.)
  if(A.individual && A.individual.length){
    html += `<h4 class="awards-section-title">${isPlayoff?'POSTSEASON':'REGULAR-SEASON'} AWARDS</h4>
      <div class="award-cards">`;
    A.individual.forEach(a=>{
      const p = a.c, isUser = p && p.team===seasonUserTeam();
      const stats = formatAwardStats(p, a.title);
      const shimmer = (p?.ovr||0) >= 90 ? 'shimmer' : (p?.ovr||0) >= 85 ? 'vibrant' : '';
      const icon = getAwardIcon(a.title);
      html += `
        <div class="award-card ${a.title.toLowerCase().replace(/\s+/g,'-')} ${isUser?'user':''} ${shimmer}" style="${isUser?'border-color:var(--team-primary);':''}">
          <div class="award-icon">${icon}</div>
          <div class="award-title">${a.title.toUpperCase()}</div>
          <div class="award-recipient">
            <div class="recipient-main">
              <span class="recipient-name ${shimmer}">${p?.name||a.coach||a.team}</span>
              ${p?`<span class="recipient-team" style="color:${getTeamColor(p.team)}">${TEAM_ABBR[p.team]}</span>`:`<span class="recipient-team">${a.team}</span>`}
            </div>
            <div class="recipient-stats">${stats}</div>
            ${p && p.ovr>=90 ? `<div class="award-shimmer-badge">90+</div>` : ''}
          </div>
        </div>
      `;
    });
    html += `</div>`;
  }
  
  // All-Pro Teams (grid cards)
  if(A.allPro){
    ['first','second'].forEach(team=>{
      if(A.allPro[team] && A.allPro[team].length){
        html += `<h4 class="awards-section-title">${team.toUpperCase()}-TEAM ALL-PRO</h4>
          <div class="allpro-grid">`;
        A.allPro[team].forEach(p=>{
          const isUser = p.team===seasonUserTeam();
          const shimmer = (p.ovr||0) >= 90 ? 'shimmer' : (p.ovr||0) >= 85 ? 'vibrant' : '';
          html += `
            <div class="ap-card ${isUser?'user':''} ${shimmer}" style="${isUser?'border-color:var(--team-primary);':''}">
              <div class="ap-avatar" style="background:${getTeamColor(p.team)}">${p.name.charAt(0)}</div>
              <div class="ap-name ${shimmer}">${p.name}</div>
              <div class="ap-meta">${LABELS[p.group]||p.group} · ${TEAM_ABBR[p.team]} · ${p.ovr}</div>
            </div>
          `;
        });
        html += `</div>`;
      });
    });
  }
  
  // Pro Bowl (conference grids)
  if(A.proBowl){
    ['AFC','NFC'].forEach(conf=>{
      if(A.proBowl[conf] && A.proBowl[conf].length){
        html += `<h4 class="awards-section-title">PRO BOWL — ${conf}</h4>
          <div class="pb-grid">`;
        A.proBowl[conf].forEach(p=>{
          const isUser = p.team===seasonUserTeam();
          const shimmer = (p.ovr||0) >= 90 ? 'shimmer' : (p.ovr||0) >= 85 ? 'vibrant' : '';
          html += `
            <div class="pb-card ${shimmer}" style="${isUser?'border-color:var(--team-primary);':''}">
              <div class="pb-avatar" style="background:${getTeamColor(p.team)}">${p.name.charAt(0)}</div>
              <div class="pb-name ${(p.ovr||0)>=90?'shimmer':''}">${p.name}</div>
              <div class="pb-meta">${LABELS[p.group]||p.group} · ${TEAM_ABBR[p.team]} · ${p.ovr}</div>
            </div>
          `;
        });
        html += `</div>`;
      });
    });
  }
  
  // User Summary
  if(A.userSummary){
    const us = A.userSummary;
    let usHtml = `<div class="awards-user-summary">`;
    if(us.ap1 || us.ap2 || us.pb || us.mine.length){
      usHtml += `<p class="awards-user-text"><b>${seasonUserTeam()}</b>: ${us.ap1} First-Team All-Pro, ${us.ap2} Second-Team All-Pro, ${us.pb} Pro Bowl selection${us.pb===1?'':'s'}${us.mine.length?` — ${us.mine.join('; ')}`:''}.</p>`;
    } else {
      usHtml += `<p class="awards-user-text">No ${seasonUserTeam()} players earned league-wide honors this year.</p>`;
    }
    usHtml += `</div>`;
    html += usHtml;
  }
  
  html += `</div>`;
  return html;
}
```

## FORMAT AWARD STATS HELPER

```javascript
function formatAwardStats(p, title){
  if(!p) return '';
  const s = p.st || {};
  const G = p.group;
  if(G==='QB') return `<span class="stat-box">QB</span><span class="stat-box">${s.cmp}/${s.att}</span><span class="stat-box passy">${s.passYds} [PassY]</span><span class="stat-box td">${s.passTD} [TD]</span><span class="stat-box int">${s.int} [INT]</span><span class="stat-box rating shimmer">${(s.rating||0).toFixed(1)} [20-0]</span><span class="stat-box rushy">${s.rushYds} [RushY]</span><span class="stat-box sk">${s.sk} [SK]</span><span class="stat-box qbr">${(s.qbr||0).toFixed(1)} [QBR]</span><span class="stat-box wins">${s.w||0} [W]</span>`;
  if(G==='RB') return `<span class="stat-box">RB</span><span class="stat-box">${s.car} car</span><span class="stat-box rushy">${s.rushYds} [RushY]</span><span class="stat-box td">${s.rushTD} [TD]</span>${s.recYds>150?`<span class="stat-box">${s.recYds} [RecY]</span>`:''}`;
  if(G==='WR'||G==='TE') return `<span class="stat-box">${LABELS[p.group]}</span><span class="stat-box">${s.rec}/${s.tgt}</span><span class="stat-box">${s.recYds} [RecY]</span><span class="stat-box td">${s.recTD} [TD]</span>`;
  if(G==='K') return `<span class="stat-box">K</span><span class="stat-box">${s.fg} FG</span><span class="stat-box">${s.xp} XP</span>`;
  const bits = [];
  if(s.sack) bits.push(`<span class="stat-box">${s.sack} [Sack]</span>`);
  if(s.dint) bits.push(`<span class="stat-box">${s.dint} [INT]</span>`);
  if(s.tfl) bits.push(`<span class="stat-box">${s.tfl} [TFL]</span>`);
  if(s.pd) bits.push(`<span class="stat-box">${s.pd} [PD]</span>`);
  if(s.ff) bits.push(`<span class="stat-box">${s.ff} [FF]</span>`);
  if(s.fr) bits.push(`<span class="stat-box">${s.fr} [FR]</span>`);
  if(s.tfl) bits.push(`<span class="stat-box">${s.tfl} [TFL]</span>`);
  if(s.pd) bits.push(`<span class="stat-box">${s.pd} [PD]</span>`);
  bits.push(`<span class="stat-box">${s.tkl||0} [TKL]</span>`);
  return bits.join('');
}
```

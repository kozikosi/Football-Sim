### 1. New `pickCritiqueVisualHTML()` Function

```javascript
function pickCritiqueVisualHTML(){
  if(typeof draftSim==='undefined' || !draftSim || !draftSim.log) return '';
  const rows = pickCritique(); 
  if(!rows.length) return '';
  
  // Verdict classification
  const verdict = r=> r.regret<=0.05 ? {label:'BEST', icon:'🏆', cls:'best'} 
    : r.regret<=0.2 ? {label:'GOOD', icon:'✅', cls:'good'} 
    : r.regret<=0.45 ? {label:'VALUE', icon:'⚠️', cls:'value'} 
    : {label:'COSTLY', icon:'❌', cls:'costly'};
  
  // Group by round
  const byRound = {};
  rows.forEach(r=>{
    (byRound[r.round] = byRound[r.round] || []).push(r);
  });
  
  // Summary stats
  const total = rows.length;
  const best = rows.filter(r=>r.regret<=0.05).length;
  const good = rows.filter(r=>r.regret<=0.2).length;
  const value = rows.filter(r=>r.regret>0.2 && r.regret<=0.45).length;
  const costly = rows.filter(r=>r.regret>0.45).length;
  const totalLeft = rows.reduce((a,r)=>a+r.regret, 0).toFixed(1);
  
  // Insights
  const insights = [];
  if(costly > 0) insights.push({icon:'❌', text:`${costly} costly miss${costly===1?'':'es'}: ${rows.filter(r=>r.regret>0.45).map(r=>`Round ${r.round} ${r.player[0]}`).join(', ')}`, cls:'costly'});
  if(value > 0) insights.push({icon:'⚠️', text:`Left ${rows.filter(r=>r.regret>0.2).reduce((a,r)=>a+r.regret,0).toFixed(1)} wins on board`, cls:'value'});
  if(good > 0) insights.push({icon:'✅', text:`${good} best available pick${good===1?'':'s'}`, cls:'good'});
  
  // Early round low-value spots
  const lowEarly = rows.filter(r=>r.round<=3 && r.v<0.35);
  if(lowEarly.length) insights.push({icon:'💡', text:`Early capital on low-impact spots: ${lowEarly.map(r=>`R${r.round} ${r.player[0]} (${slotName(r.slot)})`).join(', ')}`, cls:'info'});
  
  return `
    <div class="draft-critique-visual">
      <!-- Summary Header -->
      <div class="dcv-summary">
        <div class="dcv-stat"><span class="dcv-stat-val">${rows.length}</span><span class="dcv-stat-lbl">PICKS</span></div>
        <div class="dcv-stat"><span class="dcv-stat-val">${totalLeft}</span><span class="dcv-stat-lbl">WINS LEFT</span></div>
        <div class="dcv-verdicts">
          <span class="dcv-badge best">🏆 ${best} BEST</span>
          <span class="dcv-badge good">✅ ${good-best} GOOD</span>
          <span class="dcv-badge value">⚠️ ${value} VALUE</span>
          <span class="dcv-badge costly">❌ ${costly} COSTLY</span>
        </div>
      </div>
      
      <!-- Round Cards -->
      <div class="dcv-rounds">
        ${Object.keys(byRound).sort((a,b)=>a-b).map(rnd=>`
          <div class="dcv-round">
            <h4 class="dcv-round-title">ROUND ${rnd}</h4>
            <div class="dcv-picks">
              ${byRound[rnd].map(r=>{
                const v = verdict(r);
                const natBest = r.wasBest;
                return `
                  <div class="dcv-pick ${v.cls} ${r.wasBest?'best':''}" data-round="${r.round}" data-pick="${r.pickInRound}">
                    <div class="dcp-verdict">
                      <span class="dcp-icon">${v.icon}</span>
                      <span class="dcp-label">${v.label}</span>
                    </div>
                    <div class="dcp-player">
                      <span class="dcp-name ${r.player[3]>=90?'shimmer':r.player[3]>=85?'vibrant':''}" style="color:${getTeamColor(r.player[2])}">${r.player[0]}</span>
                      <span class="dcp-pos">${POS_DISPLAY[r.player[1]]||r.player[1]}</span>
                      <span class="dcp-ovr ${r.player[3]>=90?'shimmer':r.player[3]>=85?'vibrant':''}">${r.player[3]}</span>
                    </div>
                    <div class="dcp-slot">${slotName(r.slot)}${r.flex?' <span class="dcp-flex">FLEX</span>':''}</div>
                    <div class="dcp-wins">${fmtWar(r.v)}</div>
                    <div class="dcp-alt">${r.best && !r.wasBest ? `<span class="dcp-alt-name">${r.best.p[0]}</span> <span class="dcp-alt-detail">${POS_DISPLAY[r.best.p[1]]||r.best.p[1]} ${r.best.p[3]} at ${slotName(r.best.s)} · ${fmtWar(r.best.v)}</span>` : '<span class="dcp-your-pick">your pick</span>'}</div>
                    <div class="dcp-regret">${r.regret>0.005?'−'+r.regret.toFixed(2):'0.00'}</div>
                    <div class="dcp-verdict-text ${v.cls}">${v.label}</div>
                  </div>
                `;
              }).join('')}
            </div>
          </div>
        `).join('')}
      </div>
      
      <!-- Insights Panel -->
      <div class="dcv-insights">
        <h4 class="dci-title">KEY INSIGHTS</h4>
        <div class="dci-list">
          ${insights.map(i=>`
            <div class="dci-item ${i.cls}">
              <span class="dci-icon">${i.icon}</span>
              <span class="dci-text">${i.text}</span>
            </div>
          `).join('')}
        </div>
      </div>
    </div>
  `;
}

// Helper to group by round
const byRound = {};
rows.forEach(r=> (byRound[r.round] = byRound[r.round] || []).push(r));
```

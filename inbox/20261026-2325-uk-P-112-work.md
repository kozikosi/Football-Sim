### 1. New `renderUserScheduleHTML()`

```javascript
function renderUserScheduleHTML(){
  const user = seasonUserTeam();
  const mine = seasonState.schedule.filter(g=>g.home===user||g.away===user).sort((a,b)=>a.week-b.week);
  const currentWeek = seasonState.currentWeek;
  const isPlayoffs = seasonState.phase === 'playoffs';
  
  const cards = mine.map(g=>{
    const home = g.home===user, opp = home ? g.away : g.home;
    const oppLogo = typeof zpbTeamLogo==='function' ? zpbTeamLogo(opp) : TEAM_ABBR[opp];
    const userLogo = typeof zpbTeamLogo==='function' ? zpbTeamLogo(user) : TEAM_ABBR[user];
    const userColor = getTeamColor(user);
    const oppColor = getTeamColor(opp);
    const isCurrent = g.week === currentWeek && !seasonState.weekSimulated && !isPlayoffs;
    const isBye = false; // handle bye weeks if any
    
    let scoreHTML = '', resultClass = '', shimmerClass = '';
    
    if(g.played){
      const fs = home ? g.result.homeScore : g.result.awayScore;
      const os = home ? g.result.awayScore : g.result.homeScore;
      const won = fs > os;
      const userScoreColor = userColor;
      const oppScoreColor = getTeamColor(opp);
      
      scoreHTML = `
        <div class="sched-score ${won ? 'win' : 'loss'}" style="--user-color:${userColor}; --opp-color:${getTeamColor(opp)};">
          <span class="sched-user-score" style="color:${userScoreColor}">${fs}</span>
          <span class="sched-divider">–</span>
          <span class="sched-opp-score" style="color:${oppScoreColor}">${os}</span>
          ${g.result.ot ? '<span class="sched-ot">OT</span>' : ''}
        </div>
      `;
      resultClass = won ? 'sched-win' : 'sched-loss';
    } else if(g.week === currentWeek && !seasonState.weekSimulated) {
      // Current matchup - shimmering
      scoreHTML = `
        <div class="sched-upcoming">
          <span class="sched-time">Game Time</span>
          <span class="sched-vs">vs</span>
        </div>
      `;
      shimmerClass = 'sched-current';
    } else {
      scoreHTML = `<div class="sched-pending">—</div>`;
    }
    
    const isPlayed = g.played;
    const clickAttr = isPlayed ? `onclick="toggleRecap(${g.week})" style="cursor:pointer;"` : '';
    const recapDetail = isPlayed && g.summary ? `
      <div class="sched-recap-detail" id="recap-${g.week}" style="display:none;">
        ${renderGameResultHTML(g, user)}
      </div>
    ` : '';
    
    return `
      <div class="sched-card ${resultClass} ${shimmerClass}" ${clickAttr} data-week="${g.week}">
        <div class="sched-week">WK ${g.week}</div>
        <div class="sched-matchup">
          <div class="sched-team ${home ? 'user-home' : 'user-away'}" style="--team-color:${userColor};">
            <div class="sched-logo">${typeof zpbTeamLogo==='function' ? zpbTeamLogo(user) : TEAM_ABBR[user]}</div>
            <div class="sched-atvs">${home ? 'vs' : '@'}</div>
          </div>
          <div class="sched-opponent" style="--team-color:${getTeamColor(opp)};">
            <div class="sched-logo">${typeof zpbTeamLogo==='function' ? zpbTeamLogo(opp) : TEAM_ABBR[opp]}</div>
          </div>
        </div>
        <div class="sched-result">${scoreHTML}</div>
        ${g.divisional ? '<div class="sched-div-tag">DIV</div>' : ''}
      </div>${recapDetail}
    `;
  }).join('');
  
  // Bye weeks (if any)
  // Add bye week cards where needed
  
  // Playoffs section (if in playoffs)
  let playoffHTML = '';
  if(isPlayoffs && seasonState.playoffRounds){
    playoffHTML = `
      <div class="sched-playoffs">
        <div class="sched-playoff-header">PLAYOFFS</div>
        <div class="sched-playoff-rounds">
          ${['wildcard','divisional','champ','superbowl'].map(rk=>{
            const games = seasonState.playoffRounds[rk];
            if(!games) return '';
            const userGame = games.find(g=>g.home===user||g.away===user);
            if(!userGame) return '';
            return `
              <div class="sched-playoff-round">
                <span class="sched-round-label">${ROUND_LABELS[rk]}</span>
                <div class="sched-playoff-matchup">
                  <!-- Similar card structure for playoff game -->
                </div>
              </div>
            `;
          }).join('')}
        </div>
      </div>
    `;
  }
  
  return `
    <div class="panel sched-panel" style="margin-bottom:14px;">
      <h4 style="margin-top:0;display:flex;align-items:center;gap:10px;">
        Your Schedule & Results
        <span class="narr" style="font-weight:400;">(click a played game for recap)</span>
      </h4>
      <div class="sched-scroller">
        <div class="sched-cards">${cards}</div>
      </div>
      ${playoffHTML}
    </div>
  `;
}
```

function toggleRecap(week){
  const detail = document.getElementById('recap-'+week);
  if(detail) detail.style.display = detail.style.display === 'none' ? 'block' : 'none';
}

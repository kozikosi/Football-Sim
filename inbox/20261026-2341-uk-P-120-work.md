function renderAdvancedTeamStatsHTML(){
  const S = seasonState, user = seasonUserTeam(), T = {};
  teamNames().forEach(t=>{ T[t] = {g:0, to:0, tk:0, top:0, run:0, plays:0, td:0, fg:0, tdPts:0, fgPts:0, tpa:0, tpm:0, close:0, closeW:0, ret:0}; });
  S.schedule.forEach(g=>{ if(!g.played||!g.result||!g.detail) return;
    ['home','away'].forEach(sd=>{
      const t = g[sd], x = T[t], r = g.result, st = sd==='home'?r.homeStats:r.awayStats, ost = sd==='home'?r.awayStats:r.homeStats;
      const my = sd==='home'?r.homeScore:r.awayScore, op = sd==='home'?r.awayScore:r.homeScore;
      x.g++; x.to += st.turnovers; x.tk += ost.turnovers; x.top += st.top; x.run += st.rushYards; x.plays += st.rushYards+st.passYards;
      (r[sd+'Events']||[]).forEach(e=>{ if(e.type==='TD'){ x.td++; x.tdPts += e.pts; if(e.pts===8||(e.pts===6&&e.tpTry)){ x.tpa++; if(e.pts===8) x.tpm++; } } else if(e.type==='FG'){ x.fg++; x.fgPts += 3; } });
      if(Math.abs(my-op)<=7){ x.close++; if(my>op) x.closeW++; }
      const rt = g.detail[sd].returner; if(rt) x.ret += (rt.kry||0)+(rt.pry||0);
    });
  });
  const names = teamNames().filter(t=>T[t].g>0); if(!names.length) return '';
  
  // --- SACK DATA ---
  const cn = {rush:'individual pass rush', ol:'a weak offensive line', cov:'coverage (QB held ball)'};
  const tally = (team, key)=>{ const c = {rush:0, ol:0, cov:0}; S.schedule.forEach(g=>{ if(!g.played||!g.detail) return; ['home','away'].forEach(sd=>{ if(g[sd]===team && g.detail[sd].sackCauses) g.detail[sd].sackCauses[key].forEach(x=>{ c[x] = (c[x]||0)+1; }); }); }); return c; };
  const leagueShare = key=>{ const c = {rush:0, ol:0, cov:0}; teamNames().forEach(t=>{ const x = tally(t,key); ['rush','ol','cov'].forEach(k=>{ c[k] += x[k]; }); }); const n = c.rush+c.ol+c.cov||1; return {rush:c.rush/n, ol:c.ol/n, cov:c.cov/n, n}; };
  const made = tally(user,'made'), taken = tally(user,'taken'), lgm = leagueShare('made');
  const nm = made.rush+made.ol+made.cov, nt = taken.rush+taken.ol+taken.cov;
  const pct = (v,n)=> n? Math.round(v/n*100)+'%' : '—';
  const mine = teamPlayerStats(user);
  const dl = mine.filter(r=>r.group==='DL').sort((a,b)=>(b.st.sack||0)-(a.st.sack||0));
  const dlRows = dl.map(r=>{ const gp = Math.max(r.gp,1), e = r.st.skE||0, o = r.st.skO||0, c = r.st.skC||0, tot = e+o+c||1;
    const opp = (r.st.opOL||0)/gp, lgOL = S.league.olPass;
    return `<tr class="${r.ovr>=90?'shimmer':r.ovr>=85?'vibrant':''}"><td>${r.name} <span class="narr">${r.ovr} OVR</span></td><td class="score">${r.st.sack||0}</td><td class="score">${e}</td><td class="score">${o}</td><td class="score">${c}</td><td class="score">${opp?opp.toFixed(1):'—'}<span class="narr"> (lg ${lgOL.toFixed(1)})</span></td><td class="narr">${verdictSack(e/tot, o/tot, opp-lgOL)}</td></tr>`; }).join('');
  const ol = mine.filter(r=>r.group==='OL').sort((a,b)=>(b.st.sa||0)-(a.st.sa||0));
  const olRows = ol.map(r=>`<tr class="${r.ovr>=90?'shimmer':r.ovr>=85?'vibrant':''}"><td>${r.name} <span class="narr">${LABELS[r.slot]||r.slot} · ${r.ovr} OVR</span></td><td class="score">${r.st.sa||0}</td><td class="score">${r.st.pa||0}</td><td class="score">${r.st.pk||0}</td><td class="score">${fmtN(r.st.rbk)}</td></tr>`).join('');
  
  // --- SCORING METRICS ---
  const metrics = [
    ['Turnover margin per game', x=>(x.tk-x.to)/x.g, v=>(v>=0?'+':'')+v.toFixed(2), 'takeaways minus giveaways'],
    ['Time of possession', x=>x.top/x.g, v=>Math.floor(v/60)+':'+String(Math.round(v%60)).padStart(2,'0'), 'minutes per game'],
    ['Rushing share of yardage', x=>x.run/Math.max(x.plays,1)*100, v=>v.toFixed(0)+'%', 'how run-heavy the offense was'],
    ['Touchdowns per game', x=>x.td/x.g, v=>v.toFixed(2), 'all touchdowns, offense and defense'],
    ['Field goals per game', x=>x.fg/x.g, v=>v.toFixed(2), 'made field goals'],
    ['Share of points from TDs', x=>x.tdPts/Math.max(x.tdPts+x.fgPts,1)*100, v=>v.toFixed(0)+'%', 'the rest came from field goals'],
    ['2-Pt Conv tries', x=>x.tpa, v=>String(v), 'all season'],
    ['2-Pt Conv success', x=>x.tpa? x.tpm/x.tpa*100 : -1, v=>v<0?'—':v.toFixed(0)+'%', 'made out of tried'],
    ['Win rate in 1-score games', x=>x.close? x.closeW/x.close*100 : -1, v=>v<0?'—':v.toFixed(0)+'%', 'games decided by 7 or fewer'],
    ['Return yards per game', x=>x.ret/x.g, v=>v.toFixed(0), 'kick and punt returns']
  ];
  const rows = metrics.map(([label,f,fmt,note])=>{
    const vals = names.map(t=>({t, v:f(T[t])})).sort((a,b)=>b.v-a.v);
    const mine = vals.findIndex(x=>x.t===user)+1, lg = vals.reduce((a,x)=>a+x.v,0)/vals.length, best = vals[0], um = vals.find(x=>x.t===user).v;
    return `<tr><td>${label}<div class="narr">${note}</div></td><td class="score ${user}">${fmt(um)}</td><td class="score">${mine?('#'+mine):'—'}</td><td class="score">${fmt(lg)}</td><td>${nick(best.t)} <span class="narr">${fmt(best.v)}</span></td></tr>`;
  }).join('');
  const lgTry = names.reduce((a,t)=>a+T[t].tpa,0), lgMade = names.reduce((a,t)=>a+T[t].tpm,0);
  
  // --- SACK BREAKDOWN ---
  const lead = nm ? `Of the ${nm} sacks your defense recorded, ${pct(made.rush,nm)} came from individual pass-rush wins, ${pct(made.ol,nm)} because the opposing line broke down, and ${pct(made.cov,nm)} because coverage made the quarterback hold the ball (league average: ${Math.round(lgm.rush*100)}% / ${Math.round(lgm.ol*100)}% / ${Math.round(lgm.cov*100)}%).` : '';
  const lead2 = nt ? `Of the ${nt} sacks your offense took, ${pct(taken.rush,nt)} were beaten pass rushers, ${pct(taken.ol,nt)} were on your line, and ${pct(taken.cov,nt)} were coverage sacks (the receivers did not get open).` : '';
  
  // --- AGGRESSIVENESS ---
  const aggr = (S.aggr||{})[user]||'balanced';
  
  // --- RENDER ---
  return `
    <details class="panel adv-stats-merged" open>
      <summary style="cursor:pointer;font-weight:700;">ADVANCED TEAM STATS</summary>
      
      <!-- Scoring & Possession -->
      <section class="adv-section">
        <h4 class="adv-section-title">SCORING & POSSESSION</h4>
        <table class="adv-table">
          <thead><tr><th>Stat</th><th class="user">You</th><th>Rank</th><th>Lg Avg</th><th>League Best</th></tr></thead>
          <tbody>${rows}</tbody>
        </table>
        <p class="adv-note">League-wide: ${lgTry} 2PT tries, ${lgMade} made (${lgTry?Math.round(lgMade/lgTry*100):0}%). Aggressiveness: <b>${AGGR_INFO[aggr].label}</b>.</p>
      </section>
      
      <!-- Sack Breakdown -->
      <section class="adv-section">
        <h4 class="adv-section-title">SACKS: WHAT MADE THEM HAPPEN</h4>
        <p class="adv-narr">A sack has three causes: a rusher beats his blocker, the opposing line breaks down, or coverage holds the QB. Here's the breakdown.</p>
        
        <div class="sack-breakdown">
          <div class="sack-side">
            <h5>YOUR DEFENSE (<b>${nm}</b> sacks)</h5>
            <div class="sack-pie">
              <div class="sack-bar"><span class="sack-label">Own rush</span><span class="sack-val">${pct(made.rush,nm)}</span><div class="sack-bar-bg"><div class="sack-bar-fill" style="width:${pct(made.rush,nm)};background:var(--gold)"></div></div></div>
              <div class="sack-bar"><span class="sack-label">Weak line</span><span class="sack-val">${pct(made.ol,nm)}</span><div class="sack-bar-bg"><div class="sack-bar-fill" style="width:${pct(made.ol,nm)};background:var(--bad)"></div></div></div>
              <div class="sack-bar"><span class="sack-label">Coverage</span><span class="sack-val">${pct(made.cov,nm)}</span><div class="sack-bar-bg"><div class="sack-bar-fill" style="width:${pct(made.cov,nm)};background:var(--accent)"></div></div></div>
            </div>
            <p class="sack-note">League: ${Math.round(lgm.rush*100)}% / ${Math.round(lgm.ol*100)}% / ${Math.round(lgm.cov*100)}%</p>
          </div>
          
          <div class="sack-side">
            <h5>YOUR OFFENSE (<b>${nt}</b> taken)</h5>
            <div class="sack-pie">
              <div class="sack-bar"><span class="sack-label">Beaten</span><span class="sack-val">${pct(taken.rush,nt)}</span><div class="sack-bar-bg"><div class="sack-bar-fill" style="width:${pct(taken.rush,nt)};background:var(--bad)"></div></div></div>
              <div class="sack-bar"><span class="sack-label">Your line</span><span class="sack-val">${pct(taken.ol,nt)}</span><div class="sack-bar-bg"><div class="sack-bar-fill" style="width:${pct(taken.ol,nt)};background:var(--bad)"></div></div></div>

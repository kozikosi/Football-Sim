function renderPlayItem(p, g){
  if(p.type==='turnover'){
    const isInt = /interception/i.test(p.t);

    <span class="gr-play-icon">${icon}</span>
    <span class="gr-play-text">${formatPlayText(e, tm)}</span>
    <span class="gr-play-score">${formatScoreAtPlay(p, g)}</span>
  </div>`;
}

### C. `formatPlayText` / `formatScoreAtPlay` (fix F10)

```javascript
function formatPlayText(e, team){
  if(e.type==='TD') return `${team} — touchdown (${e.pts} pts)`;
  if(e.type==='FG') return `${team} — field goal (${e.pts} pts)`;
  if(e.type==='SAF') return `${team} — safety (${e.pts} pts)`;
  if(e.type==='OTH') return `${team} — score (${e.pts} pts)`;
  return `${team} — ${e.type} (${e.pts} pts)`;
}
function formatScoreAtPlay(p, g){
  // running score after this play's quarter is not tracked per-event; show quarter score
  const r = g.result;
  const h = (r.homeQ && r.homeQ[p.q]) || r.homeScore;
  const a = (r.awayQ && r.awayQ[p.q]) || r.awayScore;
  return `${a}–${h}`;
}
```
(Adjust ordering to away–home to match the box score; or home–away. Keep consistent with the box score's `away`/`home` rows.)

const fs=require('fs');
const w64=fs.readFileSync('/tmp/w64.js','utf8');
const w62=fs.readFileSync('/tmp/w62.js','utf8');
function extractBody(src, fn) {
  const i=src.indexOf('function '+fn+'(');
  if(i<0) return null;
  let depth=0, inStr=false, esc=false, strChar='';
  for(let j=i;j<src.length;j++) {
    const c=src[j];
    if(inStr) { if(esc) esc=false; else if(c==='\\\\') esc=true; else if(c===strChar) inStr=false; continue; }
    if(c==='\"' || c===\"'\" || c==='`') { inStr=true; strChar=c; continue; }
    if(c==='{') depth++;
    else if(c==='}') { depth--; if(depth===0) return src.slice(i, j+1); }
  }
  return null;
}
const crypto=require('crypto');

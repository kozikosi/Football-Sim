- defect (shown): Hero display is toggled by `setActiveTab` (`display: none`/`''`) synchronously mid-fade, so hero disappears instantly when leaving Draft (no fade-out) and appears instantly when returning (no fade-in) unless opacity is pre-set to 0 and animated. JS (drop viewSeason/tabSeasonBtn; use real call sites):
```javascript
const FADE_MS = 1600;
function fadeSwap(viewId, btnId) {
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) { setActiveTab(viewId, btnId); return; }
  const cur = document.querySelector('.view.active');
  const next = document.getElementById(viewId);
  const hero = document.querySelector('.hero');
  if (!next || cur === next) { setActiveTab(viewId, btnId); return; }
  if (cur) cur.classList.add('fade-out');
  if (hero && hero.style.display !== 'none') hero.classList.add('fade-out');
  setTimeout(() => {
    setActiveTab(viewId, btnId);
    const nv = document.getElementById(viewId);
    if (nv) nv.classList.add('fade-in');
    if (hero && hero.style.display !== 'none') hero.classList.add('fade-in');
    setTimeout(() => {
      if (cur) cur.classList.remove('fade-out');
      if (nv) nv.classList.remove('fade-in');
      if (hero) hero.classList.remove('fade-out','fade-in');
    }, FADE_MS + 80);
  }, FADE_MS);
}
```

```javascript
// renderIdentityPanel: bars + rating (not just bars)
function renderIdentityPanel(team, title, open, isUser){
  const S = seasonState, rt = S.unitRatings[team], rk = S.unitRanks[team];
  const rows = Object.keys(UNIT_INFO).map(u=>{
    const r = rk[u], pct = Math.max(0, (33 - r) / 32 * 100);
    const col = r <= 8 ? 'var(--gold)' : r >= 25 ? 'var(--bad)' : 'var(--accent)';
    return `<div class="identity-bar-row">
      <span class="identity-unit">${UNIT_INFO[u].label}</span>
      <span class="identity-rating">${rt[u].toFixed(1)}</span>
      <div class="identity-bar"><div class="identity-bar-fill" style="width:${pct}%;background:${col};"></div></div>
      <span class="identity-rank ${r<=8?'top':r>=25?'bottom':''}">#${r}</span>
    </div>`;
  }).join('');
  const paras = identityParagraphs(team, isUser).map(t=>`<p class="detail-item">${t}</p>`).join('');
  return `<details class="panel" style="margin-bottom:14px;" ${open?'open':''}>
    <summary style="cursor:pointer;font-weight:700;">${title} <span class="narr" style="font-weight:400;">(${team})</span></summary>
    <div style="margin-top:8px;">${paras}</div>
    <div class="identity-bars">${rows}</div>
  </details>`;
}
```

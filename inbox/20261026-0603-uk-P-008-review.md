function renderLineupTableHTML(slotmapLeft, slotmapRight, isUserLeft, isUserRight){
 let rows = '';
 DEPTH_CHART_GROUPS.forEach(([gname, gcls, slots])=>{
  rows += `<tr><td colspan="3" class="narr" style="text-transform:uppercase;letter-spacing:.05em;font-size:0.7rem;padding-top:12px;">${gname}</td></tr>`;
  slots.forEach(s=>{
   const l = lineupCell(slotmapLeft, s), r = lineupCell(slotmapRight, s);
   const lBetter = l.val > r.val, rBetter = r.val > l.val;
   const lClass = isUserLeft ? 'user' : 'opp', rClass = isUserRight ? 'user' : 'opp';
   rows += `<tr>
    <td class="${lClass} ${lBetter?'better':''}" style="${lBetter?'font-weight:700;color:var(--gold);':''}">${lBetter?'<span class="better-marker" style="color:var(--gold);margin-right:4px;font-size:0.7rem;">▲</span>':''}${l.text}</td>
    <td class="narr" style="text-align:center;">${LABELS[s]||s}</td>
    <td class="${rClass} ${rBetter?'better':''}" style="${rBetter?'font-weight:700;color:var(--gold);':''}text-align:right;">${r.text}${rBetter?'<span class="better-marker" style="color:var(--gold);margin-left:4px;font-size:0.7rem;">▲</span>':''}</td>
   </tr>`;
  });
 });
 return `<table><tbody>${rows}</tbody></table><p class="narr" style="margin:6px 0 0;">Ratings shown are effective ratings — players flexed out of position carry their penalty.</p>`;
}

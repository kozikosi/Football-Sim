### 2. Cinematic Sequence JS

```javascript
// After panel.innerHTML = h, add:
playDraftCinematic();

function playDraftCinematic(){
  const textEl = document.getElementById('dcCinematicText');
  const actionsEl = document.getElementById('dcActions');
  const cinematic20 = document.getElementById('cinematic20');
  const startBtn = document.getElementById('startSeasonFromDraftBtn');
  const c20Text = document.getElementById('cinematic20');
  
  if(!textEl) return;
  
  const lines = [
    {text: 'Play the season.', duration: 2000, hold: 1500},      // faster
    {text: 'Coach your team to the Super Bowl.', duration: 2000, hold: 1500}, // faster
    {text: 'Can you go 20-0?', duration: 3000, hold: 3000}       // lingering ≥3s, twice the hold
  ];
  
  let currentLine = 0;
  
  function showLine(){
    if(currentLine >= lines.length){
      // All lines shown - reveal actions
      textEl.style.opacity = '0';
      setTimeout(() => {
        textEl.style.display = 'none';
        actionsEl.style.display = 'flex';
        actionsEl.style.flexDirection = 'column';
        actionsEl.style.alignItems = 'center';
        actionsEl.style.gap = '12px';
        actionsEl.style.opacity = '0';
        actionsEl.style.transform = 'translateY(20px)';
        actionsEl.style.transition = 'opacity 0.8s ease, transform 0.8s ease';
        
        // Start 20-0 shimmer
        if(cinematic20) {
          cinematic20.style.animation = 'textShimmer 2.5s ease-in-out infinite';
        }
        
        // Animate in
        requestAnimationFrame(() => {
          actionsEl.style.opacity = '1';
          actionsEl.style.transform = 'translateY(0)';
        });
        
        // 20-0 text lingers ≥3s, then slow fade out
        setTimeout(() => {
          if(cinematic20) {
            cinematic20.style.transition = 'opacity 5s ease-out';
            cinematic20.style.opacity = '0';
          }
        }, 3000);
      }, 500);
      return;
    }
    
    const line = lines[currentLine];
    textEl.innerHTML = `<span class="cinematic-line foil-shimmer gold">${line.text}</span>`;
    textEl.style.opacity = '0';
    textEl.style.transform = 'translateY(20px)';
    textEl.style.transition = 'opacity 0.5s ease, transform 0.5s ease';
    
    requestAnimationFrame(() => {
      textEl.style.opacity = '1';
      textEl.style.transform = 'translateY(0)';
    });
    
    setTimeout(() => {
      textEl.style.opacity = '0';
      textEl.style.transform = 'translateY(-20px)';
      setTimeout(() => {
        currentLine++;
        showLine();
      }, 500);
    }, line.duration + line.hold);
  }
  
  // Start cinematic
  textEl.style.textAlign = 'center';
  textEl.style.fontFamily = "'Cinzel Decorative', serif";
  textEl.style.fontSize = '1.5rem';
  textEl.style.fontWeight = '700';
  textEl.style.lineHeight = '1.4';
  textEl.style.color = 'var(--gold)';
  textEl.style.maxWidth = '800px';
  textEl.style.margin = '0 auto 24px';
  textEl.style.minHeight = '80px';
  textEl.style.display = 'flex';
  textEl.style.alignItems = 'center';
  textEl.style.justifyContent = 'center';
  
  showLine();
  
  // Start Season button handler
  startBtn?.addEventListener('click', () => {
    startSeason(); 
    renderSeasonTab(document.getElementById('draftFinalResults').parentElement?.querySelector('.panel') || document.body);
  });
  
  // NEW DRAFT tab handler
  document.getElementById('resultsNewDraftBtn')?.addEventListener('click', () => {
    resultsView = 'newdraft';
    renderDraftResults();
  });
  
  // 100 SEASONS tab handler
  document.getElementById('resultsHundredBtn')?.addEventListener('click', () => {
    resultsView = 'hundred';
    renderDraftResults();
  });
}
```

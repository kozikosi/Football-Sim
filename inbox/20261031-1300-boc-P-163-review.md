Two shimmer keyframes with different names and mechanics is inconsistent. Recommend reusing `textShimmer` (or the project's standard) for the star, not inventing a new one. Also: the work's CSS has no `prefers-reduced-motion` rule for the new `shimmer` animation — it won't be disabled. Add it.

function starsFaceoffHTML(g){
  const teamColor = team => (typeof getTeamColor === 'function' ? getTeamColor(team) : 'var(--gold,#ffc43c)');
  const card = (team, players) => {
    const palette = teamColor(team);
    const isUser = team === seasonUserTeam();
    if (!players.length) {
      return `<div class="playoff-star-card" style="padding:8px 12px;background:var(--panel2);border:1px solid var(--border);border-radius:8px;margin-bottom:8px;">
        <div class="narr" style="color:var(--sub);font-size:0.85rem;">No 85+ starters for ${nick(team)}</div>
      </div>`;
    }
    return players.map(p => {
      const is90 = p.ovr >= 90;
      const star = is90 ? `<span class="playoff-star" style="color:${palette};font-size:1.1em;animation:textShimmer 2.5s ease-in-out infinite;">★</span> ` : '';
      const posLabel = LABELS[p.slot] || p.slot;
      return `<div class="playoff-star-card" style="padding:10px 12px;background:var(--panel2);border:1px solid var(--border);border-radius:8px;margin-bottom:8px;display:flex;align-items:center;gap:10px;">

      </div>`;
    }).join('');
  };

};

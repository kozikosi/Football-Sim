### Fix 1: Back-to-season Glitch

**Root cause:** The back button doesn't know where the season host is. It needs to navigate to the season tab properly.

**Solution:** Change the back button to use `setActiveTab('viewDraft','tabDraftBtn')` then trigger the season view, OR pass the correct season host reference.

**Better approach:** The draft complete panel already has a "Season" tab. The back button should switch to that tab.

**Code change at line 6135:**
```javascript
// OLD:
const back = document.getElementById('back100Btn'); if(back) back.addEventListener('click', ()=>renderSeasonTab(container));

// NEW:
const back = document.getElementById('back100Btn'); 
if(back) back.addEventListener('click', ()=>{
  // Switch to draft tab, then activate Season sub-tab
  setActiveTab('viewDraft','tabDraftBtn');
  // The draft complete panel's tab system will handle showing the Season view
  resultsView = 'season';
  renderDraftResults();
});
```

**Even better:** The draft complete panel already has tab buttons (`resultsSeasonBtn`, etc.). Just click the Season tab programmatically:
```javascript
const back = document.getElementById('back100Btn'); 
if(back) back.addEventListener('click', ()=>{
  document.getElementById('resultsSeasonBtn')?.click();
});
```

This is the cleanest fix — it uses the existing tab system.

### Helper: SVG Histogram (Aligned Axes)

```javascript
function renderHistogramSVG(hist, maxH, userAvg) {
  const wins = Object.keys(hist).map(k=>parseInt(k)).sort((a,b)=>a-b);
  const minW = Math.min(...wins), maxW = Math.max(...wins);
  const barH = 18, gap = 4, chartH = (maxW - minW + 1) * (barH + gap);
  const maxBarW = 300;
  
  let svg = `<svg width="100%" height="${chartH}" viewBox="0 0 ${maxBarW + 80} ${chartH}" style="max-width:100%;">`;
  wins.forEach((w, idx) => {
    const count = hist[w] || 0;
    const pct = maxH ? count / maxH : 0;
    const barW = pct * maxBarW;
    const isUser = Math.round(userAvg) === w;
    const y = idx * (barH + gap);
    svg += `
      <g transform="translate(60, ${y})">
        <text x="-8" y="${barH/2 + 4}" text-anchor="end" font-size="10" fill="var(--sub)">${w}</text>
        <rect x="0" y="0" width="${barW}" height="${barH}" fill="${isUser ? 'var(--gold)' : 'var(--accent)'}" rx="2"/>
        ${count ? `<text x="${barW + 6}" y="${barH/2 + 4}" font-size="10" fill="var(--text)">${count}</text>` : ''}
      </g>`;
  });
  svg += `</svg>`;
  return svg;
}
```

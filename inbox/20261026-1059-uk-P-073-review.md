// Delegated click on numeric cells in draft table
document.getElementById('draftTable2')?.addEventListener('click', e => {
  const cell = e.target.closest('td[data-sort]');
  if (!cell) return;
  const k = cell.dataset.sort;
  const sel = document.getElementById('draftSort2');
  if (sel.value === k) draftUI.sortAsc = !draftUI.sortAsc;
  else { sel.value = k; draftUI.sortAsc = ['name','pos','team'].includes(k); }
  rer();
});

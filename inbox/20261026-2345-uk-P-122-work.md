**Refined Detection:**
```javascript
function isClutchMoment(game, quarter, timeRemaining, scoreDiff){
  const isQ4 = quarter === 4;
  const lateQ4 = timeRemaining <= 600; // 10 minutes = 600 seconds
  const close = Math.abs(scoreDiff) <= 7;
  const isPlayoff = game.isPlayoff;
  
  return isQ4 && lateQ4 && close;
}
```

| **Playoff Clutch** | Same + playoff | Stronger pulse, screen shake, "PLAYOFF CLUTCH" | — |
| **Super Bowl Clutch** | Super Bowl + clutch | Full screen gold flash, "CHAMPIONSHIP CLUTCH" | — |

**Implementation in `animateGame`:**
```javascript
function checkClutchMoment(game, q, time, homeScore, awayScore){
  const diff = Math.abs(homeScore - awayScore);
  if(q !== 4) return null;
  if(time > 600) return null; // 10 minutes = 600s
  if(diff > 7) return null;
  
  const isPlayoff = game.isPlayoff;
  const isSB = game.roundLabel === 'Super Bowl';
  
  return {
    isClutch: true,
    intensity: isSB ? 'championship' : isPlayoff ? 'playoff' : 'regular',
    message: isSB ? 'CHAMPIONSHIP CLUTCH' : isPlayoff ? 'PLAYOFF CLUTCH' : 'CLUTCH'
  };
}
```

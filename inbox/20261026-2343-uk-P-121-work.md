```javascript
function initSIWScrollFade(){
  const cards = document.querySelectorAll('.siw-card');
  const observer = new IntersectionObserver((entries)=>{
    entries.forEach(entry=>{
      if(entry.isIntersecting){
        entry.target.classList.add('siw-visible');
      } else {
        entry.target.classList.remove('siw-visible');
      }
    });
  }, {threshold: 0.15, rootMargin: '0px 0px -50px 0px'});
  
  cards.forEach(card=>observer.observe(card));
}

// Call after rendering
setTimeout(initSIWScrollFade, 0);
```

```javascript
function renderWhyOffenseVisual(F){
  const {S, user, gp, T} = F, lg = S.league, names = teamNames(), R = S.unitRatings[user];
  const games = teamPlayedGames(user).filter(g=>g.played && g.result);
  if(!games.length) return '';
  
  const neut = t=> matchupModel(S.unitRatings[t], lg, lg, {off:NEUTRAL_OFF, def:DEFAULT_DEF}).structPts;
  const nv = names.map(t=>neut(t)), avgN = mean_(nv), myNraw = neut(user), rankN = nv.filter(v=>v>myNraw+1e-9).length + 1;
  const lgAct = mean_(names.map(t=>T[t].pf/Math.max(gp,1))), myN = lgAct + (myNraw - avgN);
  
  let withSched = 0, withPlans = 0, actual = 0, homeAdj = 0;
  games.forEach(g=>{
    const side = g.home===user ? 'home' : 'away', ot = side==='home' ? g.away : g.home, hf = side==='home' ? HOME_FIELD : 0;
    withSched += matchupModel(R, S.unitRatings[ot], lg, {off:NEUTRAL_OFF, def:DEFAULT_DEF}).structPts + hf;
    withPlans += g.result.exp[side].structPts + hf;
    actual += side==='home' ? g.result.homeScore : g.result.awayScore; homeAdj += hf;
  });
  const n = games.length; withSched/=n; withPlans/=n; actual/=n; homeAdj/=n;
  const schedFx = withSched - homeAdj - myNraw, planFx = withPlans - withSched, gap = actual - myN, luck = gap - schedFx - planFx;
  
  return `
    <div class="why-waterfall offense">
      <h5>WHY OFFENSE ${gap>=0?'BETTER':'WORSE'} THAN RATINGS</h5>
      <div class="waterfall">
        <div class="wf-step"><span class="wf-label">RATINGS PROJ</span><span class="wf-val">${myN.toFixed(1)}</span></div>
        <div class="wf-arrow">→</div>
        <div class="wf-step ${schedFx>=0?'pos':'neg'}"><span class="wf-label">SCHEDULE</span><span class="wf-val">${schedFx>=0?'+':''}${schedFx.toFixed(1)}</span></div>
        <div class="wf-arrow">→</div>
        <div class="wf-step ${planFx>=0?'pos':'neg'}"><span class="wf-label">GAME PLANS</span><span class="wf-val">${planFx>=0?'+':''}${planFx.toFixed(1)}</span></div>
        <div class="wf-arrow">→</div>
        <div class="wf-step ${luck>=0?'pos':'neg'}"><span class="wf-label">LUCK</span><span class="wf-val">${luck>=0?'+':''}${luck.toFixed(1)}</span></div>
        <div class="wf-arrow">→</div>
        <div class="wf-final"><span class="wf-label">ACTUAL</span><span class="wf-val">${actual.toFixed(1)}</span></div>
      </div>
      <p class="why-note">${sg(Math.abs(planFx))>=sg(Math.abs(luck))?'Coaching (plans) drove the gap':'Variance drove the gap'}</p>
    </div>
  `;
}
```

function advancePlayoffRound(){
  const rk = seasonState.currentPlayoffRound;
  const games = seasonState.playoffRounds[rk];
  const user = seasonUserTeam();
  
  // Check if user's team is still alive
  const userAlive = games.some(g=>g.home===user || g.away===user);
  
  if(!userAlive){
    // USER ELIMINATED → Simulate rest instantly, go to complete
    simulateRemainingPlayoffs();
    seasonState.phase = 'complete';
    renderSeasonTab(container); // Will call renderSeasonCompleteView
    return;
  }
  
  // User still alive - normal flow
  seasonState.playoffRoundSimulated = false;
  const order = ['wildcard','divisional','champ','superbowl'];
  const nextIdx = order.indexOf(rk) + 1;
  if(nextIdx < order.length){
    seasonState.currentPlayoffRound = order[nextIdx];
    if(nextIdx === 1) seasonState.playoffRounds.divisional = buildDivisionalRound(seasonState.seeds, seasonState.playoffRounds.wildcard);
    else if(nextIdx === 2) seasonState.playoffRounds.champ = buildConfChampRound(seasonState.playoffRounds.divisional);
    else if(nextIdx === 3) seasonState.playoffRounds.superbowl = buildSuperBowl(seasonState.playoffRounds.champ);
  } else {
    // Super Bowl done
    const sb = seasonState.playoffRounds.superbowl[0];
    seasonState.champion = sb.result.homeScore > sb.result.awayScore ? sb.home : sb.away;
    seasonState.phase = 'complete';
  }
}

function simulateRemainingPlayoffs(){
  const order = ['wildcard','divisional','champ','superbowl'];
  const currentIdx = order.indexOf(seasonState.currentPlayoffRound);
  
  for(let i = currentIdx; i < order.length; i++){
    const rk = order[i];
    const games = seasonState.playoffRounds[rk];
    games.forEach(g=>{
      g.roundLabel = ROUND_LABELS[rk];
      playGame(g, true);
      const hw = g.result.homeScore > g.result.awayScore;
      const w = hw ? g.home : g.away, l = hw ? g.away : g.home;
      g.summary = generateGameSummary(g, rk==='superbowl' ? {champion:true} : {nextRound: NEXT_ROUND_LABEL[rk]});
      // Update reach tracking for 100-season stats if needed
    });
    
    if(rk !== 'superbowl'){
      if(rk === 'wildcard') seasonState.playoffRounds.divisional = buildDivisionalRound(seasonState.seeds, seasonState.playoffRounds.wildcard);
      else if(rk === 'divisional') seasonState.playoffRounds.champ = buildConfChampRound(seasonState.playoffRounds.divisional);
      else if(rk === 'champ') seasonState.playoffRounds.superbowl = buildSuperBowl(seasonState.playoffRounds.champ);
    }
  }
  
  // Set champion
  const sb = seasonState.playoffRounds.superbowl[0];
  seasonState.champion = sb.result.homeScore > sb.result.awayScore ? sb.home : sb.away;
  seasonState.awardsPO = computePlayoffAwards();
}

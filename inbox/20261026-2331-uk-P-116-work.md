function renderStarsGameBallHTML(g){
  const d = g.detail, r = g.result;
  const mvp = d.mvp;
  const isUserGame = g.home===seasonUserTeam() || g.away===seasonUserTeam();
  
  // Pick top 4 impact players (excluding OL)

  allPlayers.sort((a,b)=>b.impact-a.impact);
  const stars = allPlayers.slice(0,4);

  const gameBall = mvp;
  
  return `

              </div>
              <div class="gr-star-line">${formatPlayerLine(p)}</div>
            </div>

        `).join('')}
      </div>
      
      <div class="gr-game-ball ${mvp.name===gameBallPlayer?'user':''}">
        <span class="gr-game-ball-icon">🏆</span>
        <span class="gr-game-ball-text">GAME BALL</span>

      </div>

}

function formatPlayerLine(p){
  const s = p.st;
  if(p.group==='QB') return `${s.cmp}/${s.att}, ${s.passYds} yds, ${s.passTD} TD${s.int?`, ${s.int} INT`:''}${s.rushYds>=20?`, ${s.rushYds} rush`:''}`;
  if(p.group==='RB') return `${s.car} car, ${s.rushYds} yds, ${s.rushTD} TD${s.recYds>=30?`, ${s.recYds} rec`:''}`;
  if(p.group==='WR'||p.group==='TE') return `${s.rec}/${s.tgt}, ${s.recYds} yds, ${s.recTD} TD`;
  if(p.group==='K') return `${s.fg} FG, ${s.xp} XP`;
  const bits = [];
  if(s.sack) bits.push(`${s.sack} sack${s.sack===1?'':'s'}`);
  if(s.dint) bits.push(`${s.dint} INT${s.dint===1?'':'s'}`);
  if(s.tfl) bits.push(`${s.tfl} TFL`);
  if(s.pd) bits.push(`${s.pd} PD`);
  if(s.ff) bits.push(`${s.ff} FF`);
  if(s.fr) bits.push(`${s.fr} FR`);
  return bits.join(', ') + `, ${s.tkl||0} tkl`;
}

function renderKeyPlaysHTML(g){
  const r = g.result;
  const allPlays = [];
  
  // Scoring plays
  ['home','away'].forEach(sd=>{
    (r[sd+'Events']||[]).forEach(e=>{
      allPlays.push({q:e.q, t:e.t, side:sd, e, type:'score'});
    });
  });
  
  // Turnovers
  const tur = turnoverParagraph(g, false, true);
  tur.items.forEach(it=>{
    allPlays.push({q:it.q||1, t:0, side:'turnover', txt:it.t, type:'turnover'});
  });
  
  // Sort chronologically
  allPlays.sort((a,b)=>a.q-b.q || a.t-b.t);
  
  return `
    <div class="gr-keyplays">
      <h4 class="gr-section-title">KEY PLAYS</h4>
      <div class="gr-qfilter">
        <button class="gr-qbtn active" data-q="all" onclick="filterPlays('all')">ALL</button>
        <button class="gr-qbtn" data-q="1" onclick="filterPlays(1)">Q1</button>
        <button class="gr-qbtn" data-q="2" onclick="filterPlays(2)">Q2</button>
        <button class="gr-qbtn" data-q="3" onclick="filterPlays(3)">Q3</button>
        <button class="gr-qbtn" data-q="4" onclick="filterPlays(4)">Q4</button>
        <button class="gr-qbtn" data-q="5" onclick="filterPlays(5)">OT</button>
      </div>
      <div class="gr-plays-list" id="grPlaysList">
        ${allPlays.map(p=>renderPlayItem(p)).join('')}
      </div>
    </div>
  `;
}

function renderPlayItem(p){
  if(p.type==='turnover'){
    return `<div class="gr-play turnover" data-q="${p.q}">
      <span class="gr-play-badge turnover">${p.txt.includes('interception')?'🛡️ INT':'🛡️ FUM'}</span>
      <span class="gr-play-text">${p.txt}</span>
    </div>`;
  }
  const sd = p.side, e = p.e;
  const tm = sd==='home' ? g.home : g.away;
  const isScore = e.pts>0;
  let icon = '';
  if(e.type==='TD') icon = '🏈';
  else if(e.type==='FG') icon = '🥅';
  else if(e.type==='SAF') icon = '🛡️';
  
  return `<div class="gr-play ${isScore?'score':''}" data-q="${p.q}">
    <span class="gr-play-q">Q${p.q}</span>
    <span class="gr-play-icon">${icon}</span>
    <span class="gr-play-text">${formatPlayText(p)}</span>
    <span class="gr-play-score">${formatScoreAtPlay(p)}</span>
  </div>`;
}

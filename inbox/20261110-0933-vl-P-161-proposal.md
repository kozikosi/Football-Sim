function renderStandingsHTML(){
 const standings = computeStandings(seasonState.schedule, teamNames());
 const user = seasonUserTeam();
 const userDiv = divisionOfTeam(user);
 let html = `<div class="panel" style="margin-bottom:14px;"><h4 style="margin-top:0;">STANDINGS</h4><div class="draft-columns" style="grid-template-columns:1fr 1fr;">`;
 Object.keys(CONFERENCES).forEach(conf=>{
  html += `<h3 style="grid-column: span 2; margin:0 0 8px;">${conf}</h3>`;
  CONFERENCES[conf].forEach(div=>{
   const rows = divisionStandings(div, standings, seasonState.schedule);
   const isUserDiv = div === userDiv;
   html += `<div class="divtable"${isUserDiv?' style="grid-column: span 2"':''}><h5>${div}</h5><table><thead><tr><th>W-L</th><th>Team</th><th>PF/G</th><th>PA/G</th><th>Diff</th></tr></thead><tbody>`;
   rows.forEach(r=>{
    const gp = r.wins+r.losses+r.ties;
    html += `<tr class="${r.team===user?'user-row':''}"><td class="score">${r.wins}-${r.losses}${r.ties?'-'+r.ties:''}</td><td>${r.team}</td><td class="narr">${gp?(r.pf/gp).toFixed(1):'—'}</td><td class="narr">${gp?(r.pa/gp).toFixed(1):'—'}</td><td class="narr">${r.diff>=0?'+':''}${r.diff}</td></tr>`;
   });
   html += `</tbody></table></div>`;
  });
 });
 return html + `</div></div>`;
}

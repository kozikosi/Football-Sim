<script id="zpb-storage-js">
// ============ Z-POST-BLUE STORAGE · PHASE 0 (w57) ============
// The Architect's contract (inbox/20261101-2100, branch architect/engine-and-seats):
//   { kind:'draft'|'week'|'season', schemaVersion:int, engineVersion:string, dataVersion:string,
//     slot:'A'|'B'|'C', savedAt:ISO string, payload }
// The envelope is the ONLY thing that crosses this boundary. The payload stays opaque — this
// module validates the envelope alone and never looks inside. Nothing here imports the engine
// (no colors/src/**, no store.ts — the §4 files are untouched). v1 is localStorage behind an
// async facade, no compression (the assessment's §B3 numbers hold). IndexedDB stays a clean
// swap: every byte goes through the injected KV. Screen call sites are a SEPARATE commit;
// ?storage=1 mounts the design showcase only.
window.ZPBStorage = (function(){
 'use strict';
 var SCHEMA = 1;                                   // this build writes schemaVersion 1
 var SLOTS = ['A','B','C'];
 var KINDS = ['draft','week','season'];
 var DEPLOYED = 'https://kozikosi.github.io/Football-Sim/'; // the brag link's only home (never a local path)
 var P = 'zpb:st:';                                // key namespace
 var SUPPORTED = { draft:1, week:1, season:1 };    // forward-only: a higher version loads as newer-version, untouched

 function kCur(s){ return P+'slot:'+s; }
 function kBak(s){ return P+'slot:'+s+':bak'; }
 function kTmp(s){ return P+'slot:'+s+':tmp'; }
 function kDel(s){ return P+'slot:'+s+':del'; }
 var K_PROFILE = P+'profile', K_SETTINGS = P+'settings', K_STATS = P+'stats', K_APPLIED = P+'stats:applied';

 // ---- the KV seam (localStorage by default; the test rig injects a Map) ----
 function defaultKV(){
  return {
   name:'localStorage',
   getItem:function(k){ try{ return window.localStorage.getItem(k); }catch(e){ return null; } },
   setItem:function(k,v){ window.localStorage.setItem(k,v); },
   removeItem:function(k){ try{ window.localStorage.removeItem(k); }catch(e){} },
   available:function(){
    try{ var t=P+'::probe'; window.localStorage.setItem(t,'1'); window.localStorage.removeItem(t); return true; }
    catch(e){ return false; }
   }
  };
 }
 var kv = defaultKV();
 function available(){ try{ return !!kv && !!kv.available && kv.available(); }catch(e){ return false; } }

 // ---- envelope law: the WHOLE of payload law is that there is none ----
 function isInt(n){ return typeof n==='number' && isFinite(n) && Math.floor(n)===n; }
 function validEnvelope(e){
  if(!e || typeof e!=='object' || Array.isArray(e)) return 'envelope';
  if(KINDS.indexOf(e.kind)<0) return 'kind';
  if(!isInt(e.schemaVersion) || e.schemaVersion<1) return 'schemaVersion';
  if(typeof e.engineVersion!=='string' || !e.engineVersion) return 'engineVersion';
  if(typeof e.dataVersion!=='string' || !e.dataVersion) return 'dataVersion';
  if(SLOTS.indexOf(e.slot)<0) return 'slot';
  if(typeof e.savedAt!=='string' || isNaN(Date.parse(e.savedAt))) return 'savedAt';
  if(!('payload' in e)) return 'payload';
  return null;
 }
 function clone(e){ return JSON.parse(JSON.stringify(e)); }

 // ---- the typed load result: ok | empty | corrupt | newer-version | unavailable ----
 function loadRaw(key){
  if(!available()) return { status:'unavailable' };
  var raw = null;
  try{ raw = kv.getItem(key); }catch(e){ return { status:'unavailable' }; }
  if(raw==null) return { status:'empty' };
  var e = null;
  try{ e = JSON.parse(raw); }catch(err){ return { status:'corrupt' }; }
  var bad = validEnvelope(e);
  if(bad) return { status:'corrupt', why:bad };
  if((SUPPORTED[e.kind]||0) < e.schemaVersion) return { status:'newer-version', envelope:e };
  return { status:'ok', envelope:e };
 }

 function readMeta(e){
  // list-level metadata straight off the envelope header — the payload is never opened
  return { kind:e.kind, schemaVersion:e.schemaVersion, engineVersion:e.engineVersion,
           dataVersion:e.dataVersion, savedAt:e.savedAt, slot:e.slot };
 }

 // ---- quota pruning: the slot's own backup first, then its soft-deleted copy.
 //      Never a completed season slot, never stats, never profile. If only protected
 //      bytes remain, refuse the write and say so — a run is never silently evicted. ----
 function pruneFor(slot){
  var freed = [];
  try{ kv.removeItem(kBak(slot)); freed.push('backup:'+slot); }catch(e){}
  try{ kv.removeItem(kDel(slot)); freed.push('trash:'+slot); }catch(e){}
  return freed;
 }

 function save(slot, envelope){
  return new Promise(function(resolve){
   var bad = validEnvelope(envelope);
   if(bad){ resolve({ ok:false, reason:'invalid-envelope', field:bad }); return; }
   if(envelope.slot!==slot){ resolve({ ok:false, reason:'slot-mismatch' }); return; }
   if(!available()){ resolve({ ok:false, reason:'unavailable' }); return; }
   var raw = JSON.stringify(envelope);
   var attempt = function(withPrune){
    try{
     kv.setItem(kTmp(slot), raw);
     var check = JSON.parse(kv.getItem(kTmp(slot)) || 'null');
     if(validEnvelope(check)){ try{ kv.removeItem(kTmp(slot)); }catch(e2){} return { ok:false, reason:'tmp-readback' }; }
     try{ var prev = kv.getItem(kCur(slot)); if(prev!=null) kv.setItem(kBak(slot), prev); }catch(e3){}
     kv.setItem(kCur(slot), raw);
     try{ kv.removeItem(kTmp(slot)); }catch(e4){}
     return { ok:true, bytes:raw.length, pruned:withPrune||[] };
    }catch(err){
     try{ kv.removeItem(kTmp(slot)); }catch(e5){}
     return { ok:false, reason:(String(err && ((err.name||'')+' '+(err.message||''))||'').toLowerCase().indexOf('quota')>=0?'quota':'write-failed'), error:String(err) };
    }
   };
   var r = attempt(null);
   if(!r.ok && r.reason==='quota'){
    // only QUOTA prunes + retries (the one-deep backup and the trash are the expendable bytes).
    // a plain interrupted write is reported — the tmp key already protected the current bytes,
    // and the next save heals; silently retrying would mask the very fault the rig must see.
    var freed = pruneFor(slot);
    r = attempt(freed);
    if(r.ok){ r.pruned = freed; } else { r.pruned = freed; }
   }
   if(r.ok) try{ externalNote(slot); }catch(e6){}
   resolve(r);
  });
 }

 function load(slot){
  return new Promise(function(resolve){
   var r = loadRaw(kCur(slot));
   if(r.status==='ok'){ resolve({ status:'ok', envelope:r.envelope, meta:readMeta(r.envelope), usedBackup:false }); return; }
   if(r.status==='corrupt'){
    // the one-deep backup protects a failed write — if it validates, it steps forward, and the UI says so
    var b = loadRaw(kBak(slot));
    if(b.status==='ok'){
     try{ kv.setItem(kCur(slot), JSON.stringify(b.envelope)); }catch(e){}
     resolve({ status:'ok', envelope:b.envelope, meta:readMeta(b.envelope), usedBackup:true, repairedFrom:'backup' }); return;
    }
    resolve({ status:'corrupt', usedBackup:false, why:r.why }); return;
   }
   if(r.status==='newer-version'){ resolve({ status:'newer-version', envelope:r.envelope, meta:readMeta(r.envelope) }); return; }
   resolve(r); // empty | unavailable
  });
 }

 function list(){
  return new Promise(function(resolve){
   if(!available()){ resolve({ status:'unavailable', slots:[] }); return; }
   var out = [];
   var finish = function(){
    var next = function(slot){
     if(slot>=SLOTS.length){ resolve({ status:'ok', slots:out }); return; }
     var s = SLOTS[slot], r = loadRaw(kCur(s));
     if(r.status==='ok') out.push(Object.assign({ slot:s, empty:false }, readMeta(r.envelope)));
     else if(r.status==='newer-version') out.push({ slot:s, empty:false, kind:r.envelope.kind, savedAt:r.envelope.savedAt, newer:true, schemaVersion:r.envelope.schemaVersion });
     else if(r.status==='corrupt') out.push({ slot:s, empty:false, kind:null, corrupt:true });
     else out.push({ slot:s, empty:true });
     next(slot+1);
    };
    next(0);
   };
   finish();
  });
 }

 function remove(slot){ // soft: the run steps into the trash, restorable until the next save
  return new Promise(function(resolve){
   if(!available()){ resolve({ ok:false, reason:'unavailable' }); return; }
   try{
    var cur = kv.getItem(kCur(slot));
    if(cur==null){ resolve({ ok:true, gone:true }); return; }
    kv.setItem(kDel(slot), cur);
    kv.removeItem(kCur(slot));
    kv.removeItem(kBak(slot));
    externalNote(slot);
    resolve({ ok:true, soft:true });
   }catch(e){ resolve({ ok:false, reason:'write-failed', error:String(e) }); }
  });
 }
 function restore(slot){ // undo the soft delete
  return new Promise(function(resolve){
   var t = kv.getItem(kDel(slot));
   if(t==null){ resolve({ ok:false, reason:'nothing-to-restore' }); return; }
   var r = null; try{ r = JSON.parse(t); }catch(e){}
   if(validEnvelope(r)){ resolve({ ok:false, reason:'corrupt' }); return; }
   try{ kv.setItem(kCur(slot), t); kv.removeItem(kDel(slot)); externalNote(slot); resolve({ ok:true }); }
   catch(e){ resolve({ ok:false, reason:'write-failed' }); }
  });
 }

 function migrate(){
  return new Promise(function(resolve){
   // forward-only: v1 has no older versions. The walk proves every slot parses; a future
   // migration map replaces the identity step. newer-version slots are never touched.
   list().then(function(l){
    if(l.status!=='ok'){ resolve({ status:l.status, migrated:0, kept:0 }); return; }
    var migrated = 0, kept = 0;
    l.slots.forEach(function(s){ if(s.newer) kept++; else if(!s.empty && !s.corrupt) migrated++; });
    resolve({ status:'ok', migrated:migrated, kept:kept });
   });
  });
 }

 // ---- multi-tab: a storage event on our namespace means another tab holds the pen ----
 function externalNote(){
  try{
   window.dispatchEvent(new CustomEvent('zpb:storage-external', { detail:{ at:Date.now() } }));
  }catch(e){}
 }
 if(typeof window!=='undefined' && window.addEventListener){
  window.addEventListener('storage', function(ev){
   if(ev && ev.key && String(ev.key).indexOf(P)===0){
    try{ window.dispatchEvent(new CustomEvent('zpb:storage-external', { detail:{ key:ev.key, at:Date.now(), remote:true } })); }catch(e){}
   }
  });
 }

 // ---- the global stores (keyed by franchise, never by slot) ----
 function readJSON(key, fallback){
  var raw = kv.getItem(key);
  if(raw==null) return fallback;
  try{ return JSON.parse(raw); }catch(e){ return fallback; }
 }
 function writeJSON(key, val){ try{ kv.setItem(key, JSON.stringify(val)); return true; }catch(e){ return false; } }

 var SETTINGS_DEFAULT = { cinematicsReduced:false, bragHints:true };
 var settingsApi = {
  read:function(){ if(!available()) return Object.assign({}, SETTINGS_DEFAULT); return Object.assign({}, SETTINGS_DEFAULT, readJSON(K_SETTINGS, {})); },
  write:function(patch){ var cur = settingsApi.read(); var next = Object.assign({}, cur, patch||{}); return writeJSON(K_SETTINGS, next) ? next : null; }
 };

 var PROFILE_DEFAULT = { createdAt:null, lastTeam:null, lastPlayedAt:null, achievements:[] };
 var profileApi = {
  read:function(){ if(!available()) return Object.assign({}, PROFILE_DEFAULT); return Object.assign({}, PROFILE_DEFAULT, readJSON(K_PROFILE, {})); },
  touch:function(patch){ var cur = profileApi.read(); var next = Object.assign({}, cur, patch||{}); if(!next.createdAt) next.createdAt = new Date().toISOString(); writeJSON(K_PROFILE, next); return next; },
  award:function(a){ // achievements survive slot deletion AND overwrite — they live here, outside the slots
   var cur = profileApi.read();
   var id = a && a.id; if(!id) return cur;
   if(!(cur.achievements||[]).some(function(x){ return x.id===id; })){
    cur.achievements = (cur.achievements||[]).concat([{ id:a.id, label:a.label, at:new Date().toISOString(), team:a.team||null }]);
    writeJSON(K_PROFILE, cur);
   }
   return cur;
  }
 };

 // ---- stats: overall AND by franchise. The elimination trigger lives with the WRITER (the
 //      engine side sets payload.statsRecorded at the moment of elimination); this module
 //      refuses a double count on its own ledger too (belt and braces, the assessment's way). ----
 var RESULT_RANK = { missed:0, wildcard:1, divisional:2, champ:3, 'sb-loss':4, champion:5 };
 var RESULT_PTS  = { 0:0, 1:15, 2:25, 3:40, 4:60, 5:100 };
 function blankFrac(){ return { seasons:0, wins:0, losses:0, ties:0, titles:0, sb:0, conf:0, trips:0, undefeated:0,
   bestRecord:null, bestResult:null }; }
 function statsDefault(){ return { schema:1, overall:blankFrac(), franchises:{} }; }
 function betterRecord(a, b){ // more wins, then fewer losses; null-proof
  if(!b) return true; if(!a) return false;
  if(a.w!==b.w) return a.w>b.w; if((a.l||0)!==(b.l||0)) return (a.l||0)<(b.l||0); return (a.t||0)>(b.t||0);
 }
 function betterResult(a, b){ return (a==null?false:(b==null?true:(RESULT_RANK[a]||0)>(RESULT_RANK[b]||0))); }
 function legendOf(f){
  // THE FRANCHISE LEGEND — integer, explainable: a title is worth 120, the best run the
  // franchise ever had is worth its ladder value (0 missed · 15 wildcard · 25 divisional ·
  // 40 conference · 60 a Super Bowl lost · 100 a Super Bowl won), two per career win, and
  // 30 for every undefeated regular season.
  if(!f) return 0;
  var rp = RESULT_PTS[f.bestResult!=null?(RESULT_RANK[f.bestResult]||0):0] || 0;
  return (f.titles||0)*120 + rp + (f.wins||0)*2 + (f.undefeated||0)*30;
 }
 function applySeason(slot, envelope){
  // consumes ONLY the season payload's stats fields (contract §3): payload.user {team, w, l, t,
  // result, undefeated?}, payload.statsRecorded. Anything missing → applied:false, never guessed.
  return new Promise(function(resolve){
   if(!available()){ resolve({ applied:false, reason:'unavailable' }); return; }
   if(!envelope || envelope.kind!=='season'){ resolve({ applied:false, reason:'kind' }); return; }
   var p = envelope.payload || {};
   var u = p.user;
   if(!u || typeof u.team!=='string' || !isInt(u.w) || !isInt(u.l)){ resolve({ applied:false, reason:'payload-shape' }); return; }
   if(p.statsRecorded===true){ resolve({ applied:false, reason:'already-recorded' }); return; }
   var ledger = readJSON(K_APPLIED, {});
   var key = envelope.slot+':'+envelope.savedAt+':'+u.team;
   if(ledger[key]){ resolve({ applied:false, reason:'ledger' }); return; }
   var st = readJSON(K_STATS, null) || statsDefault();
   if(st.schema!==1){ resolve({ applied:false, reason:'schema' }); return; }
   var t = u.team;
   var fr = st.franchises[t] || blankFrac();
   var rec = { w:u.w, l:u.l, t:u.t||0, year:u.year||null };
   var res = RESULT_RANK[u.result]!=null ? u.result : 'missed';
   fr.seasons++; fr.wins+=rec.w; fr.losses+=rec.l; fr.ties+=rec.t;
   if(res==='champion'){ fr.titles++; fr.sb++; }
   if(res==='sb-loss'){ fr.sb++; }
   if(res==='champ'||res==='sb-loss'||res==='champion'){ fr.conf++; }
   if(res!=='missed'){ fr.trips++; }
   if(u.undefeated===true){ fr.undefeated++; }
   if(betterRecord(rec, fr.bestRecord)) fr.bestRecord = rec;
   if(betterResult(res, fr.bestResult)) fr.bestResult = res;
   st.franchises[t] = fr;
   var o = st.overall;
   o.seasons++; o.wins+=rec.w; o.losses+=rec.l; o.ties+=rec.t;
   if(res==='champion'){ o.titles++; o.sb++; }
   if(res==='sb-loss'){ o.sb++; }
   if(res==='champ'||res==='sb-loss'||res==='champion'){ o.conf++; }
   if(res!=='missed'){ o.trips++; }
   if(u.undefeated===true){ o.undefeated++; }
   if(betterRecord(rec, o.bestRecord)) o.bestRecord = Object.assign({ team:t }, rec);
   if(betterResult(res, o.bestResult)) o.bestResult = res;
   o.legend = legendOf(o);
   fr.legend = legendOf(fr);
   ledger[key] = true;
   var wrote = writeJSON(K_STATS, st) && writeJSON(K_APPLIED, ledger);
   if(!wrote){ resolve({ applied:false, reason:'quota' }); return; }
   if(u.undefeated===true) profileApi.award({ id:'undefeated', label:'Undefeated regular season', team:t });
   if(res==='champion') profileApi.award({ id:'title-'+t, label:'Super Bowl champion', team:t });
   profileApi.touch({ lastTeam:t, lastPlayedAt:new Date().toISOString() });
   resolve({ applied:true, stats:st, franchise:fr, legend:fr.legend });
  });
 }

 // ---- the brag share (clipboard + toast; the deployed URL only, never a local path) ----
 function bragUrl(){
  try{
   if(location.protocol==='file:' || location.protocol==='about:') return DEPLOYED;
   if(!/^https?:$/.test(location.protocol)) return DEPLOYED;
   if(/^(localhost|127\.0\.0\.1)/.test(location.hostname)) return DEPLOYED;
   return location.origin + location.pathname;
  }catch(e){ return DEPLOYED; }
 }
 function bragText(f){ // f = {team, w, l, result, undefeated?}
  var result = f.result==='champion' ? 'won it all' : (f.result==='sb-loss' ? 'made the Super Bowl' :
    (f.result==='champ' ? 'made the conference championship' : (RESULT_RANK[f.result]>=1 ? 'made the playoffs' : 'gave it everything')));
  var rec = f.w+'-'+f.l+((f.t||0)?('-'+f.t):'');
  var undefeated = f.undefeated ? ' A perfect '+f.w+'-0 regular season.' : '';
  return 'I went '+rec+' and '+result+' with '+f.team+'.'+undefeated+' Can you go 20-0? '+bragUrl();
 }
 function bragCopy(text){
  return new Promise(function(resolve){
   var done = function(okk){ resolve(okk); };
   try{
    if(navigator.clipboard && navigator.clipboard.writeText){
     navigator.clipboard.writeText(text).then(function(){ done(true); }, function(){ done(legacyCopy(text)); });
     return;
    }
   }catch(e){}
   done(legacyCopy(text));
  });
 }
 function legacyCopy(text){
  try{
   var ta = document.createElement('textarea');
   ta.value = text; ta.setAttribute('readonly','');
   ta.style.cssText = 'position:fixed;left:-9999px;top:0;';
   document.body.appendChild(ta); ta.select();
   var okk = document.execCommand('copy');
   ta.remove(); return !!okk;
  }catch(e){ return false; }
 }

 // ---- the test rig: a Map-backed KV with fault hooks (interrupted write, corruption, quota,
 //      unavailable, two tabs) — the battery drives this, the real screens never see it ----
 function makeTestKV(){
  var store = new Map();
  return {
   name:'map-kv', store:store, failNext:0, quotaMode:false, down:false,
   available:function(){ return !this.down; },
   getItem:function(k){ if(this.down) throw new Error('unavailable'); return store.has(k)?store.get(k):null; },
   setItem:function(k,v){
    if(this.down) throw new Error('unavailable');
    if(this.quotaMode) throw new Error('QuotaExceededError');
    if(this.failNext>0){ this.failNext--; throw new Error('SimulatedInterruptedWrite'); }
    store.set(k,String(v));
   },
   removeItem:function(k){ if(this.down) throw new Error('unavailable'); store.delete(k); }
  };
 }
 function _test(){
  var rig = { pass:0, fail:0, lines:[] };
  var ok = function(c, label){ rig.lines.push((c?'PASS  ':'FAIL  ')+label); c?rig.pass++:rig.fail++; };
  var base = makeTestKV();
  var savedKv = kv; kv = base;
  try{
   var env = function(slot, kind, extra){ return Object.assign({ kind:kind||'season', schemaVersion:1, engineVersion:'w56-ts-a1',
     dataVersion:'csv-77a2f1', slot:slot, savedAt:new Date().toISOString(), payload:Object.assign({ stub:true }, extra||{}) }); };
   // T1 · round trip
   var p = null;
   save('A', env('A')).then(function(r1){
    ok(r1.ok===true, 'T1 save ok'); return load('A');
   }).then(function(l1){
    ok(l1.status==='ok' && l1.envelope.payload.stub===true, 'T1 load ok (payload opaque, round trip)');
    return list();
   }).then(function(l2){
    ok(l2.slots.length===3 && l2.slots[0].slot==='A' && l2.slots[0].kind==='season' && l2.slots[1].empty===true, 'T1 list reads the header only');
    // T2 · the envelope law refuses garbage
    return save('B', { kind:'nonsense', payload:{} });
   }).then(function(r2){
    ok(r2.ok===false && r2.reason==='invalid-envelope' && r2.field==='kind', 'T2 bad envelope refused');
    return load('B');
   }).then(function(l3){
    ok(l3.status==='empty', 'T2 refused write left the slot empty');
    // T3 · interrupted write: the tmp dies, the current survives, the next save heals
    base.failNext = 1;
    return save('B', env('B','draft'));
   }).then(function(r3){
    ok(r3.ok===false && r3.reason==='write-failed', 'T3 interrupted write reported');
    ok(base.store.get(kCur('B'))==null, 'T3 current untouched by the interrupted write');
    return save('B', env('B','draft'));
   }).then(function(r4){
    ok(r4.ok===true, 'T3 the next save heals');
    return save('B', env('B','draft')); // the second save lays the one-deep backup for T4
   }).then(function(r4b){
    ok(r4b.ok===true, 'T3 a second save lays the backup');
    // T4 · corruption with a good backup heals from the backup
    base.store.set(kCur('B'), '{not json');
    return load('B');
   }).then(function(l4){
    ok(l4.status==='ok' && l4.usedBackup===true && l4.repairedFrom==='backup', 'T4 corrupt current restored from the one-deep backup');
    // T5 · corruption with no usable backup reports corrupt
    base.store.set(kCur('B'), '{also not'); base.store.set(kBak('B'), '{neither');
    return load('B');
   }).then(function(l5){
    ok(l5.status==='corrupt', 'T5 corrupt with no backup reads corrupt');
    // T6 · quota: the expendable bytes prune first, the run itself is never evicted
    base.store.set(kDel('B'), '{"trash":1}');
    base.quotaMode = true;
    return save('B', env('B','week'));
   }).then(function(r6){
    base.quotaMode = false;
    ok(r6.ok===false && r6.reason==='quota', 'T6 quota refused the write instead of evicting a run');
    ok((r6.pruned||[]).indexOf('trash:B')>=0, 'T6 the expendable bytes pruned first (trash, backup)');
    ok(base.store.get(kCur('B'))==='{also not', 'T6 the existing slot bytes were never the price');
    // T7 · unavailable reads typed
    base.down = true;
    return load('C');
   }).then(function(l7){
    ok(l7.status==='unavailable', 'T7 unavailable is a typed result');
    base.down = false;
    // T8 · newer-version is kept, never deleted, never faked
    return save('C', Object.assign(env('C'), { schemaVersion:2 })).then(function(r8){
     ok(r8.ok===true, 'T8 a newer envelope stores fine (storage does not judge versions)');
     return load('C');
    });
   }).then(function(l8){
    ok(l8.status==='newer-version' && l8.envelope.schemaVersion===2, 'T8 newer-version loads typed and kept');
    // T9 · soft delete + restore
    return save('A', env('A')).then(function(){ return remove('A'); }).then(function(r9){
     ok(r9.ok===true && r9.soft===true, 'T9 soft delete');
     return list();
    }).then(function(l9){
     ok(l9.slots[0].empty===true, 'T9 the soft-deleted slot reads empty in the list');
     return restore('A');
    }).then(function(r10){
     ok(r10.ok===true, 'T9 restore brings the run back');
     return load('A');
    }).then(function(l10){
     ok(l10.status==='ok', 'T9 the restored slot loads');
     // T11 · the stats elimination trigger + the double-count guard
     var season = env('A','season',{ statsRecorded:false, user:{ team:'Carolina Panthers', w:14, l:3, t:0, result:'champion', year:2026, undefeated:false } });
     return applySeason('A', season).then(function(s1){
      ok(s1.applied===true && s1.franchise.titles===1 && s1.legend===248, 'T11 the elimination moment records once (legend 120 title + 100 champion + 28 wins = 248)');
      return applySeason('A', Object.assign({}, season, { payload:Object.assign({}, season.payload, { statsRecorded:true }) }));
     }).then(function(s2){
      ok(s2.applied===false && s2.reason==='already-recorded', 'T11 the payload flag refuses the double count');
      return applySeason('A', season);
     }).then(function(s3){
      ok(s3.applied===false && s3.reason==='ledger', 'T11 the ledger refuses the replayed season too');
      // T12 · migrate is a no-op forward walk on v1
      return migrate();
     }).then(function(m){
      ok(m.status==='ok' && m.kept===1, 'T12 migrate walks forward, keeps the newer-version slot');
     });
    });
   }).catch(function(err){ ok(false, 'unexpected rig error — '+String(err).slice(0,90)); })
    .then(function(){ kv = savedKv; if(typeof window!=='undefined') window.__zpbStorageTest = rig; });
  }catch(e){ kv = savedKv; ok(false, 'rig threw — '+String(e).slice(0,90)); }
  return rig;
 }

```python
def resolve_division_ties(team_a, team_b):
    # Step 1: Head-to-Head Win Percentage
    if team_a.h2h_win_pct != team_b.h2h_win_pct:
        return team_a if team_a.h2h_win_pct > team_b.h2h_win_pct else team_b
        
    # Step 2: Division Win Percentage
    if team_a.div_win_pct != team_b.div_win_pct:
        return team_a if team_a.div_win_pct > team_b.div_win_pct else team_b
        
    # Step 3: Common Games Win Percentage (min 4 games)
    if team_a.common_win_pct != team_b.common_win_pct:
        return team_a if team_a.common_win_pct > team_b.common_win_pct else team_b
        
    # Step 4: Conference Win Percentage
    if team_a.conf_win_pct != team_b.conf_win_pct:
        return team_a if team_a.conf_win_pct > team_b.conf_win_pct else team_b
        
    # Step 5: Strength of Victory (SOV - combined win % of opponents beaten)
    if team_a.sov != team_b.sov:
        return team_a if team_a.sov > team_b.sov else team_b
        
    # Step 6: Strength of Schedule (SOS - combined win % of all opponents)
    if team_a.sos != team_b.sos:
        return team_a if team_a.sos > team_b.sos else team_b
        
    # Step 7: Best Net Points in Division / All Games
    if team_a.point_diff != team_b.point_diff:
        return team_a if team_a.point_diff > team_b.point_diff else team_b

    # Step 8: Coin Toss
    return team_a
```

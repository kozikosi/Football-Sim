// ================= Z-POST-BLUE W60 · THE NAME LADDER + THE ADVANCE RAIL =================
// w60 (Owner): "once the window size is too constrained for full names horizontally, then it
// shows full names vertically. Then if the window size is too constrained to show the names
// vertically, to then default to showing last names, and then if the screen is too compressed
// for last names, then the screen shows players' initials. No more '…' anywhere on the website."
// "SINGLE WORDS IN NAMES SHOULD BE ON THEIR OWN LINE" and "REMEMBER THAT JR. IS NOT A LAST NAME,
// IT STANDS FOR JUNIOR." The ladder measures each carrier's own box (not the viewport), caches
// the full name, and is stateless — growing the window restores the fuller form. Suffixes
// (Jr./Sr./II/III/IV/V) are stripped before the last-name step, so a suffix alone is never the
// last name. The CSS guarantees words stay whole (overflow-wrap:normal); the ellipsis paints
// are removed everywhere.
(function(){
'use strict';
 var SEL = '.zpb-lu-cell,.zpb-gname,.zpb-awname,.zpb-awme,.zpb-dz-name,.zpb-whyname,'
  + '.zpb-tree .bracket-game .zpb-trow2 em,.zpb-fillab,.zpb-pickname,.zpb-pickrow b,'
  + '.zpb-pcard-hd b,.zpb-pb-name,.zpb-pb-team,.zpb-intro-star b,td.dc-name,.zst-opp';
 var SUF = /^(?:jr|sr|ii|iii|iv|v)\.?$/i;
 var meas = null, busy = false, queued = false;
 function measurerFor(el){
  if(!meas || !meas.parentNode){
   meas = document.createElement('span');
   meas.setAttribute('aria-hidden','true');
   meas.style.cssText = 'position:fixed;left:-9999px;top:0;visibility:hidden;white-space:nowrap;z-index:-1;';
   document.body.appendChild(meas);
  }
  var cs = getComputedStyle(el);
  meas.style.font = cs.font;
  meas.style.letterSpacing = cs.letterSpacing;
  meas.style.textTransform = cs.textTransform;
  meas.style.fontVariant = cs.fontVariantCaps || cs.fontVariant;
  return meas;
 }
 function widthOf(text, el){ var m = measurerFor(el); m.textContent = text; return m.getBoundingClientRect().width; }
 function avail(el){ var w = el.clientWidth; if(!w){ var p = el.parentElement; w = p ? p.clientWidth : 0; } return w; }
 function parse(full){
  var toks = full.split(/\s+/).filter(Boolean), rest = [];
  while(toks.length && /^[0-9]+$/.test(toks[toks.length-1])) rest.unshift(toks.pop());
  return { nm: toks, rest: rest.length ? ' ' + rest.join(' ') : '' };
 }
 function longestTok(tokens){
  var best = '';
  for(var i=0;i<tokens.length;i++){
   var parts = String(tokens[i]).split('-');
   for(var j=0;j<parts.length;j++){ if(parts[j].length > best.length) best = parts[j]; }
  }
  return best;
 }
 // The pure ladder — exposed for the battery. fits(text) answers "does this text fit one line
 // in the carrier's box?" (the caller measures). Returns {text, mode}: one = single line,
 // wrap = whole-word vertical.
 function ladder(full, fits){
  var p = parse(full);
  if(!p.nm.length) return { text: full, mode: 'one' };
  if(fits(p.nm.join(' ') + p.rest)) return { text: full, mode: 'one' };
  if(fits(longestTok(p.nm))) return { text: p.nm.join(' ') + p.rest, mode: 'wrap' };
  var lt = p.nm.slice();
  while(lt.length > 1 && SUF.test(lt[lt.length-1].replace(/\./g,''))) lt.pop();
  var ln = lt[lt.length-1] || p.nm[p.nm.length-1] || '';
  if(fits(ln) || fits(longestTok([ln]))) return { text: ln + p.rest, mode: 'one' };
  var ini = p.nm.filter(function(t){ return !SUF.test(t.replace(/\./g,'')); })
    .map(function(t){ return t.charAt(0); }).join('').toUpperCase();
  return { text: (ini || ln) + p.rest, mode: 'one' };
 }
 function fit(el){
  var full = el.getAttribute('data-nmfull');
  if(!full){ full = (el.textContent || '').replace(/\s+/g,' ').trim(); if(!full) return; el.setAttribute('data-nmfull', full); }
  var w = avail(el);
  if(w <= 0) return;
  var d = ladder(full, function(txt){ return widthOf(txt, el) <= w; });
  if(el.textContent !== d.text) el.textContent = d.text;
  el.style.whiteSpace = d.mode === 'one' ? 'nowrap' : 'normal';
  el.style.overflowWrap = 'normal';
 }
 function scan(){
  if(busy) return; busy = true;
  try{
   var list = document.querySelectorAll(SEL);
   for(var i=0;i<list.length;i++) fit(list[i]);
  } finally { busy = false; }
 }
 function schedule(){
  if(queued || busy) return; queued = true;
  setTimeout(function(){ queued = false; scan(); }, 120);
 }
 try{ new MutationObserver(schedule).observe(document.body, {childList:true, subtree:true}); }catch(e){}
 window.addEventListener('resize', schedule, {passive:true});
 window.zpbNameFit = { ladder: ladder, scan: scan };
 scan();
})();

// ---- THE ADVANCE RAIL ----
// w60 (Owner): "ANY GLOWING BUTTON THAT HELP ADVANCE YOU TO THE NEXT PAGE SHOULD ALSO BE
// AVAILABLE FOR THE USER TO CLICK AT THE TOP AND BOTTOM OF EVERY SCREEN DURING THE SEASON
// (EXCEPT THE HOMEPAGE AND FOOTER PAGES). THE USER SHOULDN'T HAVE TO SCROLL BACK UP OR SCROLL
// DOWN TO ADVANCE TO THE NEXT PAGE." The rail mirrors the current season screen's advance
// button — same label, same glow, same action — fixed at the very top, so the action is
// reachable at any scroll position. No registered button on screen (homepage, draft setup,
// the footer pages) → no rail. Inside .wrap, so the cinematic lock locks it with the page.
(function(){
'use strict';
 var IDS = ['simWeekBtn','nextWeekBtn','nextRoundBtn','beginPlayoffsBtn','seasonReviewBtn','beginSeasonBtn','beginSeasonBtn2'];
 var rail = null, cur = null, queued = false;
 function findCta(){
  var view = document.getElementById('viewDraft');
  if(!view || !view.classList.contains('active')) return null;
  for(var i=0;i<IDS.length;i++){
   var b = document.getElementById(IDS[i]);
   if(b && b.offsetParent !== null) return b;
  }
  return null;
 }
 function build(b){
  var c = b.cloneNode(true);
  c.id = 'zpbAdvRailBtn';
  c.removeAttribute('data-nmfull');
  c.addEventListener('click', function(ev){ ev.preventDefault(); try{ b.click(); }catch(e){} });
  return c;
 }
 function sync(){
  var b = findCta();
  if(!b){
   if(rail) rail.style.display = 'none';
   document.body.classList.remove('zpb-advrail');
   cur = null;
   return;
  }
  if(!rail){
   var wrap = document.querySelector('.wrap');
   if(!wrap) return;
   rail = document.createElement('div');
   rail.id = 'zpbAdvRail';
   wrap.appendChild(rail);
  }
  if(!cur || cur.__src !== b || cur.textContent !== b.textContent){
   while(rail.firstChild) rail.removeChild(rail.firstChild);
   cur = build(b);
   cur.__src = b;
   rail.appendChild(cur);
  }
  rail.style.display = 'flex';
  document.body.classList.add('zpb-advrail');
 }
 function schedule(){
  if(queued) return; queued = true;
  setTimeout(function(){ queued = false; try{ sync(); }catch(e){} }, 80);
 }
 try{ new MutationObserver(schedule).observe(document.body, {childList:true, subtree:true}); }catch(e){}
 window.addEventListener('resize', schedule, {passive:true});
 window.zpbAdvRail = { sync: sync };
 sync();
})();
